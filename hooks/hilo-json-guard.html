Qué comprueba

ESTADO_PROYECTO.json lleva claves de configuracion duplicadas con valores divergentes (ADR-023)

Códigos de salida

ExitSignificado
0permitir
1AVISO (configuracion divergente; politica warn)
2BLOCK intrinseco (un _hilo/*.json que quedaria vacio o invalido, al escribirlo o al commitearlo) o config-guard en hooks.blockList

Política

Bloquea siempre: es uno de los irrenunciables (daño irreversible), no configurable.

ClaveTipoDefaultQué hace
hooks.policystring"warn"WARN-first por defecto (avisan sin bloquear); 'block' eleva todos los hooks defensivos a bloqueo. Los hooks intrinsecamente BLOCK (bash-guard, protect-files certs/keys, hilo-json-guard) no se degradan nunca.
hooks.blockListarray[]Hooks concretos elevados a BLOCK aunque policy=warn (ej. ['secret-scanner']).

// hilo-json-guard.js - Bloquea escribir o commitear un _hilo/*.json vacio o invalido (FB-007, Portal); avisa si
// ESTADO_PROYECTO.json lleva claves de configuracion duplicadas con valores divergentes (ADR-023)
// Evento: PreToolUse[Bash|Write|Edit]
// Exit 0 = permitir · Exit 1 = AVISO (configuracion divergente; politica warn) · Exit 2 = BLOCK intrinseco (un _hilo/*.json que quedaria vacio o invalido, al escribirlo o al commitearlo) o config-guard en hooks.blockList
// Version: 1.1.0
// Historial del original (condensado):
//   v1.1.0 (ADR-032, H-03): tambien en Write|Edit. La promesa "JSON del Hilo invalido bloquea siempre" solo se
//     cumplia al commitear: un Write de "{bad" a _hilo/ESTADO_PROYECTO.json pasaba con exit 0. Ahora se valida el
//     contenido que quedaria en disco (Write: el content; Edit: el fichero con el reemplazo aplicado) antes de escribir.
//   v1.0.0 (Ovillo, FB-007): un hook dejo _hilo/ESTADO_PROYECTO.json a 0 bytes Y se committeo ->
//     rompio el @import del CLAUDE.md y /mcp-sync. Intercepta 'git commit' y rechaza si algun
//     _hilo/*.json STAGED esta vacio o no parsea. Complementa writeHiloJson (defensa en profundidad).
//
// Politica: BLOCK INTRINSECO — data-loss del state file, dano concreto e inmediato;
// resolvePolicy con intrinsicBlock:true devuelve 'block' incondicionalmente.

'use strict';

const fs = require('fs');
const path = require('path');
const h = require('./lib/helpers');
const config = require('./lib/config');

const ES_HILO_JSON = /(^|[\\/])_hilo[\\/][^\\/]+\.json$/i;

function bloquear(mode, bad, verbo) {
  console.error('');
  console.error('BLOQUEADO hilo-json-guard: vas a ' + verbo + ' un _hilo/*.json vacio o invalido (data loss).');
  for (const b of bad) console.error(`  - ${b}`);
  console.error('');
  console.error('Un _hilo/*.json vacio o invalido rompe el @import del CLAUDE.md y /mcp-sync.');
  console.error(verbo === 'commitear'
    ? 'Fix: restaura el contenido (git checkout HEAD -- <archivo> o regeneralo) y re-stagea antes de commitear.'
    : 'Fix: escribe el JSON completo y valido (los hooks usan writeHiloJson de .claude/hooks/lib/helpers.js).');
  process.exit(mode === 'block' ? 2 : 1); // intrinseco: siempre 2
}

function validarTexto(raw) {
  const s = String(raw == null ? '' : raw).replace(/^﻿/, '');
  if (!s.trim()) return 'vacio / 0 bytes';
  try { JSON.parse(s); return null; } catch (e) { return 'JSON invalido: ' + (e && e.message ? e.message : e); }
}

try {
  const hookData = h.readHookInput();
  if (!hookData) process.exit(0);
  const toolInput = h.getToolInput(hookData);
  if (!toolInput) process.exit(0);

  const mode = config.resolvePolicy(config.load(), 'hilo-json-guard', { intrinsicBlock: true }); // siempre 'block'
  const repoRoot = process.env.CLAUDE_PROJECT_DIR || process.cwd();
  const tool = String(hookData.tool_name || '');
  const ti = hookData.tool_input && typeof hookData.tool_input === 'object' ? hookData.tool_input : null;

  // ---- Write | Edit: el contenido que quedaria en disco ----------------------------------------------------------
  if (tool === 'Write' || tool === 'Edit' || tool === 'MultiEdit') {
    const filePath = h.getFilePath(hookData);
    if (!filePath || !ES_HILO_JSON.test(filePath)) process.exit(0);
    let contenido = null;
    if (tool === 'Write') {
      contenido = ti && typeof ti.content === 'string' ? ti.content : (h.getNewContent(hookData) || '');
    } else {
      // Edit: aplicar el reemplazo sobre el fichero actual. Si el old_string no casa, Claude Code rechazara el
      // Edit por su cuenta; si el fichero no existe, no hay nada que proteger.
      const abs = path.isAbsolute(filePath) ? filePath : path.join(repoRoot, filePath);
      if (!fs.existsSync(abs) || !ti || typeof ti.old_string !== 'string' || typeof ti.new_string !== 'string') process.exit(0);
      const actual = fs.readFileSync(abs, 'utf8');
      if (!actual.includes(ti.old_string)) process.exit(0);
      contenido = ti.replace_all ? actual.split(ti.old_string).join(ti.new_string) : actual.replace(ti.old_string, () => ti.new_string);
    }
    const problema = validarTexto(contenido);
    if (problema) bloquear(mode, [`${filePath} (${problema})`], 'escribir');
    process.exit(0);
  }

  // ---- Bash: solo 'git commit' -----------------------------------------------------------------------------------
  let command = null;
  if (ti && ti.command) {
    command = ti.command;
  } else {
    const m = toolInput.match(/"command"\s*:\s*"((?:[^"\\]|\\.)*)"/);
    if (m) command = m[1];
  }
  if (!command) process.exit(0);

  // Solo activa en 'git commit' (no diff, log, etc.)
  if (!/\bgit\s+commit\b/i.test(command)) process.exit(0);

  // Leer staged files (added/modified). Sin lock (GIT_OPTIONAL_LOCKS=0 via gitReadOnly)
  // para no colisionar con VS/GitLens.
  const diff = h.gitReadOnly(['-c', 'core.quotepath=off', 'diff', '--cached', '--name-only', '--diff-filter=AM'], repoRoot);
  if (diff.exitCode !== 0) process.exit(0); // si git falla (repo nuevo, primer commit), no bloquear
  const staged = diff.stdOut.split(/\r?\n/).filter(Boolean);
  if (staged.length === 0) process.exit(0);

  const duranJsons = staged.filter((f) => /(^|\/)_hilo\/.*\.json$/i.test(f));
  if (duranJsons.length === 0) process.exit(0);

  const bad = [];
  for (const f of duranJsons) {
    const show = h.gitReadOnly(['show', `:${f}`], repoRoot); // contenido STAGED (indice), no el working tree
    if (show.exitCode !== 0) continue; // no se pudo leer el staged: no bloquear por esto
    const problema = validarTexto(show.stdOut);
    if (problema) bad.push(`${f} (${problema})`);
  }

  // Coherencia de la configuracion (ADR-023): solo si ESTADO_PROYECTO.json esta staged. WARN-first, elevable con
  // hooks.blockList: ['config-guard']. NUNCA intrinseco: una divergencia no es data-loss.
  if (bad.length === 0 && duranJsons.some((f) => /ESTADO_PROYECTO\.json$/i.test(f))) {
    const divergent = config.validate(repoRoot).filter((f) => f.code === 'DUP_DIVERGENTE');
    if (divergent.length) {
      const cfgMode = config.resolvePolicy(config.load(), 'config-guard');
      console.error('');
      console.error((cfgMode === 'block' ? 'BLOQUEADO' : 'AVISO') + ' config-guard: _hilo/ESTADO_PROYECTO.json tiene claves de configuracion duplicadas con valores distintos:');
      for (const f of divergent) console.error(`  - ${f.path} = ${JSON.stringify(f.valores.plana)} frente a configuracion.branching = ${JSON.stringify(f.valores.branching)}`);
      console.error('Gana configuracion.branching.* (ADR-023). Fix: ' + h.cmdEcosistema('hooks/config-check.js', '--migrate'));
      process.exit(cfgMode === 'block' ? 2 : 1);
    }
  }

  if (bad.length > 0) bloquear(mode, bad, 'commitear');

  process.exit(0);
} catch (e) {
  // Error inesperado: permitir operacion en lugar de romper el flujo.
  console.error(`hilo-json-guard.js: error inesperado, permitiendo operacion. Detalle: ${e && e.message ? e.message : e}`);
  process.exit(0);
}
