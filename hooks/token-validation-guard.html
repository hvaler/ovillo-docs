Qué comprueba

Salida: hallazgos y motivo del bloqueo por stderr, que es lo que Claude Code muestra con exit 1 y 2 (stdout solo es contexto en SessionStart y UserPromptSubmit; ADR-004 D1 enmendado por ADR-030) WARN-first: el .ps1 v1.0.0 salia siempre 0 (avisos informativos); en el port, con hallazgos resolvePolicy(cfg, 'token-validation-guard') decide exit 1 (warn, default) o exit 2 (block) segun ecosystem.config -> hooks.policy / hooks.blockList. Mismos patrones y mensajes que el .ps1 (los patrones se aplican sobre el contenido JSON-escapado del tool_input, igual que el original).

Códigos de salida

ExitSignificado
0sin hallazgos
1AVISO (politica warn)
2BLOCK (politica block)

Política

WARN-first: avisa (exit 1) por defecto y bloquea (exit 2) solo si la organización lo pone en hooks.blockList o sube hooks.policy a block.

ClaveTipoDefaultQué hace
hooks.policystring"warn"WARN-first por defecto (avisan sin bloquear); 'block' eleva todos los hooks defensivos a bloqueo. Los hooks intrinsecamente BLOCK (bash-guard, protect-files certs/keys, hilo-json-guard) no se degradan nunca.
hooks.blockListarray[]Hooks concretos elevados a BLOCK aunque policy=warn (ej. ['secret-scanner']).

#!/usr/bin/env node
// token-validation-guard.js - Verificar TokenValidationParameters
// Evento: PreToolUse[Write|Edit] (para que exit 2 pueda impedir la escritura)
// Exit 0 = sin hallazgos · Exit 1 = AVISO (politica warn) · Exit 2 = BLOCK (politica block)
// Salida: hallazgos y motivo del bloqueo por stderr, que es lo que Claude Code muestra con exit 1 y 2 (stdout solo es
//         contexto en SessionStart y UserPromptSubmit; ADR-004 D1 enmendado por ADR-030)
// Version: 1.0.0
//   WARN-first: el .ps1 v1.0.0 salia siempre 0 (avisos informativos); en el port,
//   con hallazgos resolvePolicy(cfg, 'token-validation-guard') decide exit 1 (warn,
//   default) o exit 2 (block) segun ecosystem.config -> hooks.policy / hooks.blockList.
//   Mismos patrones y mensajes que el .ps1 (los patrones se aplican sobre el contenido
//   JSON-escapado del tool_input, igual que el original).

'use strict';

const h = require('./lib/helpers');
const cfgLib = require('./lib/config');

function main() {
  const hookData = h.readHookInput();

  // Solo archivos .cs
  const filePath = h.getFilePath(hookData) || '';
  if (!/\.cs$/i.test(filePath)) return 0;

  // Contenido REAL, ya des-escapado (h.getNewContent). Casar los patrones sobre el JSON
  // serializado daba falsos negativos por construccion: una interpolacion llega como $\" y
  // ningun patron escrito contra la comilla sin escapar la ve.
  const content = h.getNewContent(hookData);
  if (!content) return 0;

  // Solo actuar si hay TokenValidationParameters
  if (!/TokenValidationParameters/i.test(content)) return 0;

  let warnings = 0;

  // === CHECKLIST TOKEN VALIDATION ===

  // ValidateIssuer
  if (/ValidateIssuer\s*=\s*false/i.test(content)) {
    console.error('SEGURIDAD: ValidateIssuer=false. Cualquier emisor seria aceptado.');
    warnings++;
  }

  // ValidateAudience
  if (/ValidateAudience\s*=\s*false/i.test(content)) {
    console.error('SEGURIDAD: ValidateAudience=false. Tokens para otras apps serian aceptados.');
    warnings++;
  }

  // ValidateLifetime
  if (/ValidateLifetime\s*=\s*false/i.test(content)) {
    console.error('SEGURIDAD: ValidateLifetime=false. Tokens expirados serian aceptados.');
    warnings++;
  }

  // ValidateIssuerSigningKey
  if (/ValidateIssuerSigningKey\s*=\s*false/i.test(content)) {
    console.error('SEGURIDAD: ValidateIssuerSigningKey=false. Tokens sin firma valida serian aceptados.');
    warnings++;
  }

  // RequireExpirationTime
  if (/RequireExpirationTime\s*=\s*false/i.test(content)) {
    console.error('SEGURIDAD: RequireExpirationTime=false. Tokens sin expiracion serian aceptados.');
    warnings++;
  }

  // Algoritmos inseguros
  if (/SecurityAlgorithms\.(HmacSha256|None)|"none"|"HS256"/i.test(content)) {
    if (/"none"/i.test(content) || /SecurityAlgorithms\.None/i.test(content)) {
      console.error("SEGURIDAD CRITICA: Algoritmo 'none' detectado. Tokens sin firma serian aceptados.");
      warnings++;
    }
  }

  // ClockSkew excesivo
  if (/ClockSkew\s*=\s*TimeSpan\.MaxValue|ClockSkew\s*=\s*TimeSpan\.FromHours/i.test(content)) {
    console.error('SEGURIDAD: ClockSkew excesivo. Maximo recomendado: 5 minutos.');
    warnings++;
  }

  if (warnings > 0) {
    const policy = cfgLib.resolvePolicy(cfgLib.load(), 'token-validation-guard');
    console.error('');
    console.error(`Se detectaron ${warnings} problema(s) en TokenValidationParameters.`);
    console.error('Referencia: Estandar del ecosistema - IdP con validacion completa obligatoria.');
    console.error('Checklist: ValidateIssuer=true, ValidateAudience=true, ValidateLifetime=true,');
    console.error('           ValidateIssuerSigningKey=true, ClockSkew<=5min.');
    console.error(`Politica: ${policy} (ecosystem.config hooks.policy)`);
    return policy === 'block' ? 2 : 1;
  }

  return 0;
}

try {
  process.exit(main());
} catch (e) {
  console.error(`token-validation-guard: error inesperado (${e.message}) - no se bloquea el flujo`);
  process.exit(0);
}
