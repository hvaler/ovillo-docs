# {{PROYECTO_NOMBRE}} — Pipeline CI + CD (Fase 2, stack Vue 3 + Vite + TypeScript)
# Ovillo cicd-architect vue @ {{TEMPLATE_STAMP}}   (sello de plantilla; /cicd-status detecta drift)
# Diferencias vs .NET (azure-pipelines.fase2.dotnet.yml.template):
#   - Stage Build: NodeTool@0 + npm ci + vue-tsc + lint + vitest + npm run build + ArchiveFiles@2
#   - TakeAppOfflineFlag: false (R10 — SPA estatica, NO usar true)
#   - Smoke: best-effort (R11 — NO throw, emite warning + diagnostico)
# Resto idéntico al template .NET (mismos stages CD on-demand R20, trigger por estrategia R21, mismas invariantes G1-G19).
# =============================================================================

name: $(date:yyyyMMdd)$(rev:.r)

trigger:
  # R21: branches segun configuracion.branching.estrategia (ESTADO_PROYECTO.json).
  #   push-directo (github-flow-simplificado / trunk-based / developer-branch) -> solo {{RAMA_BASE}}, SIN branch policy build-validation (bloquearia el push directo).
  #   con-PR (github-flow / gitflow / release-flow / oneflow / gitlab-flow) -> /cicd-init anade ramas en el marcador + RECOMIENDA branch policy para el gate en PR.
  branches:
    include:
      - {{RAMA_BASE}}
      # {{TRIGGER_BRANCHES_EXTRA}}  <- /cicd-init R21: gitflow=+develop ; release-flow/oneflow=+'release/*' ; gitlab-flow=+pre-production/+production ; push-directo=eliminar este marcador
  paths:
    include:
      - 'src/**'
      - 'src/**/*'
      - 'azure-pipelines.yml'

pr: none

variables:
  - group: {{PIPELINE_NAME}}-secrets
  # R20 (TEC-009, piloto GuiasDocentes): deploy on-demand. DeployEnv NO se declara aqui — es variable
  # de la DEFINICION (settable at queue time, default '', la pone /cicd-init FASE 2.5). Declararla en
  # el YAML la ECLIPSA: en runtime vale siempre '' y los stages de deploy quedan SIEMPRE skipped (CD
  # on-demand no dispara). /cicd-deploy --env <env> la pone al encolar; las condiciones la leen igual.
  - name: nodeVersion
    value: '22.x'                                            # Node 22 LTS default del ecosistema
  - name: workingDir
    value: 'src'
  - name: buildOutDir
    value: '{{BUILD_OUTDIR}}'                                # default 'dist' de vite.config.ts
  - name: iisVirtualApp
    value: '{{IIS_VIRTUAL_APP}}'
  - name: iisPhysicalPath         # R15: ruta fisica REAL de la vApp en disco. Vacio = C:\inetpub\wwwroot\<vApp>.
    value: '{{IIS_PHYSICAL_PATH}}'  # Rellenar si la vApp NO cuelga de wwwroot. La consumen backup-pre-deploy.yml + rollback-on-fail.yml.
  - name: backupRoot              # R15: raiz de backups pre-deploy. Override si el server no tiene unidad D:.
    value: 'C:\Backups'

stages:
  # ───────────────────────────────────────────────────────────────────────
  # STAGE 1: BUILD (Vue+Vite+TS)
  # ───────────────────────────────────────────────────────────────────────
  - stage: Build
    displayName: 'Build & Test (Vue+Vite+TS)'
    jobs:
      - job: BuildAndTest
        pool:
          name: {{BUILD_POOL}}
          demands:
            - Agent.ComputerName -equals {{BUILD_AGENT}}
        timeoutInMinutes: 30
        steps:
          # R5: NUNCA Cache@2. npm ci limpio acepta los 30-60s extra
          - task: NodeTool@0                                  # R4
            displayName: 'Install Node.js $(nodeVersion)'
            inputs:
              versionSpec: '$(nodeVersion)'

          - script: npm ci
            displayName: 'npm ci (con lockfile)'
            workingDirectory: '$(workingDir)'

          - script: npx vue-tsc --noEmit
            displayName: 'Type check (vue-tsc)'
            workingDirectory: '$(workingDir)'

          - script: npm run lint
            displayName: 'ESLint'
            workingDirectory: '$(workingDir)'
            continueOnError: true                             # warning-only

          - script: npx vitest run --coverage
            displayName: 'Tests + coverage (Vitest V8)'
            workingDirectory: '$(workingDir)'

          - task: PublishCodeCoverageResults@1                # R4 NO @2
            displayName: 'Publish coverage'
            inputs:
              codeCoverageTool: 'Cobertura'
              summaryFileLocation: '$(workingDir)/coverage/cobertura-coverage.xml'

          # R22: gate de ciberseguridad variante SPA (npm audit; hardening/SAST skipped).
          # NO tumba el Build; publica 'security-report' para el gate de los stages de deploy.
          - template: steps/security-scan-build.yml
            parameters:
              stack: spa

          - script: npm run build
            displayName: 'Vite build (production)'
            workingDirectory: '$(workingDir)'

          - task: ArchiveFiles@2                              # R4
            displayName: 'Zip $(buildOutDir)/'
            inputs:
              rootFolderOrFile: '$(workingDir)/$(buildOutDir)'
              includeRootFolder: false
              archiveType: 'zip'
              archiveFile: '$(Build.ArtifactStagingDirectory)/{{APP_NAME}}.zip'

          - task: PublishBuildArtifacts@1
            displayName: 'Publish artifact drop'
            inputs:
              PathtoPublish: '$(Build.ArtifactStagingDirectory)'
              ArtifactName: 'drop'

  # ───────────────────────────────────────────────────────────────────────
  # STAGES DeployDev / DeployDemo / DeployProd
  # ───────────────────────────────────────────────────────────────────────
  # IDENTICOS al template .NET con DOS unicas diferencias:
  #
  #   1. TakeAppOfflineFlag: false  ← R10 SPA estatica (NUNCA true)
  #   2. Usar smoke-besteffort-spa.yml (NO smoke-strict-dotnet.yml) ← R11
  #
  # El model on-demand R20 (variable DeployEnv arriba + `condition:
  # and(succeeded(), eq(variables['DeployEnv'], '<env>'))` en cada stage de deploy)
  # es IDENTICO al .NET. Push no despliega; /cicd-deploy --env <env> activa el stage.
  #
  # ROUTING (R25): una SPA Vue es una aplicacion web PUBLICA -> SIEMPRE servidores de
  # Aplicaciones. /cicd-init genera los stages de deploy con DeployTarget=<env>-app
  # (dev-app/pre-app/pro-app) y smoke best-effort a la URL de apps (dev01/demo01/intranet).
  # (No aplica el matiz publico/privado de las APIs; una SPA nunca va a Servicios.)
  #
  # El resto (pool {{BUILD_POOL}} + DeployTarget capability <env>-app, REST API download,
  # security-verdict-gate.yml R22 como PRIMER step de cada deploy (warn DEV / block PRE-PROD),
  # backup pre-deploy R15, IISWebAppDeploymentOnMachineGroup@0,
  # rollback condition:failed() R15 capa 3, group approval PROD) es identico.
  #
  # Ver: .claude/skills/cicd-architect/templates/azure-pipelines.fase2.dotnet.yml.template
  #      para la estructura completa de los stages CD.
