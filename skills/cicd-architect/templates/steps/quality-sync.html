# =============================================================================
# steps/quality-sync.yml
# Publica el Registro de Calidad (QR) del build en el Hub Ovillo para el dashboard
# de calidad (skill calidad-codigo-sync). Corre en el stage Build DESPUES de los tests,
# con continueOnError:true (NO tumba el Build: la calidad no gatea, eso es el gate de cobertura G7/R18).
#
# PRE-REQUISITO: el paso de tests debe emitir coverage en formato OpenCover ademas de
# cobertura. Usar el runsettings de la skill en el 'dotnet test':
#   --settings $(Build.SourcesDirectory)/ops/testing/coverlet.runsettings
#   (Format=cobertura,opencover). cobertura sigue alimentando el gate de cobertura (G7/R18).
#
# Variables del pipeline esperadas:
#   $(qualityProjectId) - GUID del proyecto en el hub (mismo que mcpSync.projectId)
#   $(codeSearchBase)   - base ADO Code Search del repo (el dashboard concatena el fichero)
# Secreto del Variable Group (R16, isSecret): HUB_SERVICE_KEY.
#   Emitir una vez: EXEC mcp.EmitirServiceKey @ProyectoId='<GUID>', @Label='ci-quality', ...
#
# El script Sync-QualityRecord.ps1 lo materializa /cicd-init en ops/cicd/scripts/ (el plugin no vive en
# el repo del proyecto, asi que CI no puede leer .claude/skills/).
# Auth: la service-key NO es la apiKey per-dev; es credencial de automatizacion.
# Se mapea por env (NO como argumento en linea para no exponerla en el log).
# =============================================================================
parameters:
  - name: crapThreshold
    type: number
    default: 30
  - name: coverageLineMin
    type: number
    default: 75
  - name: coverageBranchMin
    type: number
    default: 60
  - name: topHotspots
    type: number
    default: 20
  # Host del Hub ALCANZABLE DESDE EL AGENTE. Si el agente esta en una red interna, el FQDN publico puede
  # no resolver (hairpin/split-DNS): apuntar al host interno. El default de Sync-QualityRecord.ps1 vale para
  # /calidad-sync LOCAL; en CI, fijar este parametro por proyecto.
  - name: hubServerUrl
    type: string
    default: 'https://hub.example.org'

steps:
  - task: PowerShell@2
    displayName: 'Publicar QR de calidad en el Hub (best-effort, no bloquea)'
    continueOnError: true
    condition: always()        # corre aunque algun test falle: el coverage parcial sigue siendo util
    env:
      HUB_SERVICE_KEY: $(HUB_SERVICE_KEY)   # secreto VG -> env (masked); el script lo lee con -ApiKey
    inputs:
      # SIN pwsh: no todos los agentes on-premise tienen PowerShell Core.
      # Sync-QualityRecord.ps1 es #Requires -Version 5.1 -> corre en Windows PowerShell 5.1 (presente).
      targetType: inline
      script: |
        if ([string]::IsNullOrWhiteSpace($env:HUB_SERVICE_KEY)) {
          Write-Warning "HUB_SERVICE_KEY vacia en el Variable Group; QR no publicado. (R16: anadir el secreto isSecret)."
          exit 0
        }
        & "$(Build.SourcesDirectory)\ops\cicd\scripts\Sync-QualityRecord.ps1" `
          -ProjectId "$(qualityProjectId)" `
          -ApiKey    $env:HUB_SERVICE_KEY `
          -ResultsDir "$(Agent.TempDirectory)" `
          -CodeSearchBase "$(codeSearchBase)" `
          -BuildId   "$(Build.BuildNumber)" `
          -ServerUrl "${{ parameters.hubServerUrl }}" `
          -CrapThreshold ${{ parameters.crapThreshold }} `
          -CoverageLineMin ${{ parameters.coverageLineMin }} `
          -CoverageBranchMin ${{ parameters.coverageBranchMin }} `
          -TopHotspots ${{ parameters.topHotspots }}
