# =============================================================================
# steps/security-verdict-gate.yml
# R22: descarga el artefacto 'security-report' (REST, R3) y aplica el verdict.
#   mode=warn  (DeployDev)      -> vuelca el verdict como warning, NO bloquea.
#   mode=block (DeployDemo/Prod) -> throw ANTES del deploy si effectiveSummary tiene
#                                   hallazgos >= gateSeverity o secretos > 0.
# Fail-safe: si NO hay verdict valido -> en mode=block BLOQUEA (nunca asume "0 findings").
# Colocar como PRIMER step del job de deploy (antes de backup/MSDeploy).
# R2: TLS 1.2+.
# =============================================================================
parameters:
  - name: mode
    type: string
    default: block         # block | warn
  - name: gateSeverity
    type: string
    default: High          # severidad efectiva minima que bloquea

steps:
  - task: PowerShell@2
    displayName: 'R22 gate seguridad (${{ parameters.mode }})'
    env:
      SYSTEM_ACCESSTOKEN: $(System.AccessToken)
    inputs:
      targetType: inline
      script: |
        try { [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 -bor [Net.SecurityProtocolType]::Tls13 } catch { [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 }
        $mode = '${{ parameters.mode }}'
        $gate = '${{ parameters.gateSeverity }}'
        $headers = @{ Authorization = "Bearer $env:SYSTEM_ACCESSTOKEN" }
        $apiBase = "$(System.CollectionUri)$(System.TeamProject)/_apis/build/builds/$(Build.BuildId)"
        $report = $null
        try {
          $info = Invoke-RestMethod -Uri "$apiBase/artifacts?artifactName=security-report&api-version=5.0" -Headers $headers
          $zip  = Join-Path $env:AGENT_TEMPDIRECTORY 'security-report.zip'
          Invoke-WebRequest -Uri $info.resource.downloadUrl -Headers $headers -OutFile $zip
          $dest = Join-Path $env:AGENT_TEMPDIRECTORY 'security-report-dl'
          if (Test-Path $dest) { Remove-Item $dest -Recurse -Force }
          Expand-Archive -Path $zip -DestinationPath $dest -Force
          $json = Get-ChildItem -Path $dest -Recurse -Filter 'security-report.json' | Select-Object -First 1
          if ($json) { $report = Get-Content $json.FullName -Raw | ConvertFrom-Json }
        } catch { Write-Warning "No se pudo obtener el security-report: $_" }

        if (-not $report) {
          if ($mode -eq 'block') { throw "R22: no hay verdict de seguridad valido. BLOQUEO (no se asume '0 findings'). Revisar el step de Build 'R22 security scan'." }
          Write-Warning "R22: sin verdict de seguridad. mode=warn -> se continua."
          return
        }

        $eff = $report.effectiveSummary
        Write-Host "R22 verdict efectivo: Crit=$($eff.Critical) High=$($eff.High) Mod=$($eff.Moderate) Low=$($eff.Low) Secretos=$($eff.secrets)"
        $order = @{ 'Low' = 1; 'Moderate' = 2; 'High' = 3; 'Critical' = 4 }
        $gateLevel = $order[$gate]
        $blockers = 0
        foreach ($sev in 'Critical','High','Moderate','Low') {
          if ($order[$sev] -ge $gateLevel) { $blockers += [int]$eff.$sev }
        }
        $blockers += [int]$eff.secrets
        if ($blockers -gt 0) {
          $msg = "R22: $blockers hallazgo(s) >= $gate o secretos en el verdict efectivo."
          if ($mode -eq 'block') {
            throw "$msg DEPLOY BLOQUEADO. Resolver o anadir excepcion vigente en ops/testing/.security-exceptions.yml. Ver ops/cicd/SEGURIDAD_PIPELINE.md."
          }
          Write-Warning "$msg mode=warn -> se despliega igual (DEV)."
        } else {
          Write-Host "R22: verdict limpio para gate >= $gate. Continua el deploy."
        }
