# =============================================================================
# {{PROYECTO_NOMBRE}} — Workflow CI (Fase 1: build validation, sin CD)
# Ovillo cicd-architect github-dotnet-fase1 @ {{TEMPLATE_STAMP}}   (sello de plantilla; /cicd-status detecta drift)
# Reglas: cicd-runtime G1-G19 (G15 permisos minimos, G16 pin por SHA, G19 concurrency, G2 merge_group)
# Materializado por /cicd-init en .github/workflows/ci.yml
#
# Alcance (Fase 1, CI-only): compila y testea la SOLUCION COMPLETA en cada PR y en la rama base. NO despliega.
# Al promover a Fase 2, /cicd-init quita el trigger `push` de este fichero (lo construye cd-<slug>.yml).
#
# RAMAS CONDICIONALES que /cicd-init aplica al generar:
#   (a) SIN proyecto de test -> eliminar los steps 'Test' y 'Upload coverage'.
#   (b) SIN merge queue en el ruleset -> puede quitarse `merge_group`; con ella es obligatorio (G2).
# El ruleset de main exige los checks `Build` y `Security` (git-best-practices/templates/branch-ruleset.json.template).
# =============================================================================
name: CI

on:
  push:
    branches:
      - '{{RAMA_BASE}}'
      # {{TRIGGER_BRANCHES_EXTRA}}  <- /cicd-init anade aqui "- 'develop'" en el modelo branch-gated
    paths-ignore:
      - '**/*.md'
      - 'docs/**'
      - '_hilo/**'
      - '_patron/**'
  pull_request:
    branches: ['{{RAMA_BASE}}']
  merge_group:
    branches: ['{{RAMA_BASE}}']

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

env:
  DOTNET_VERSION: '{{NET_VERSION}}'
  SOLUTION_PATH: '{{SLN_PATH}}'
  TEST_FILTER: '{{TEST_FILTER}}'

jobs:
  build:
    name: Build
    runs-on: '{{RUNNER}}'
    timeout-minutes: 20
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

      - name: Setup .NET
        uses: actions/setup-dotnet@a98b56852c35b8e3190ac28c8c2271da59106c68 # v6.0.0
        with:
          dotnet-version: ${{ env.DOTNET_VERSION }}

      - name: Cache NuGet packages
        uses: actions/cache@55cc8345863c7cc4c66a329aec7e433d2d1c52a9 # v6.1.0
        with:
          path: ~/.nuget/packages
          key: nuget-${{ runner.os }}-${{ hashFiles('**/*.csproj', '**/Directory.Packages.props') }}
          restore-keys: nuget-${{ runner.os }}-

      - name: Restore
        run: dotnet restore "${{ env.SOLUTION_PATH }}"

      - name: Build
        run: dotnet build "${{ env.SOLUTION_PATH }}" --no-restore --configuration Release

      - name: Test
        shell: bash
        run: |
          args=(--no-build --configuration Release --results-directory TestResults)
          [ -n "${TEST_FILTER}" ] && args+=(--filter "${TEST_FILTER}")
          [ -f ops/testing/coverlet.runsettings ] && args+=(--settings ops/testing/coverlet.runsettings)
          dotnet test "${{ env.SOLUTION_PATH }}" "${args[@]}"

      - name: Coverage gate (G7, warn)
        if: always()
        uses: ./.github/actions/coverage-gate
        with:
          path: TestResults
          threshold-line: '70'
          threshold-branch: '60'
          mode: warn

      - name: Upload coverage
        if: github.event_name == 'pull_request'
        uses: actions/upload-artifact@043fb46d1a93c77aae656e7c1c64a875d1fc6a0a # v7.0.1
        with:
          name: coverage-${{ github.run_number }}
          path: TestResults/**/coverage.cobertura.xml
          if-no-files-found: ignore
          retention-days: 14

  security:
    name: Security
    runs-on: '{{RUNNER}}'
    timeout-minutes: 10
    needs: build
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

      - name: Setup .NET
        uses: actions/setup-dotnet@a98b56852c35b8e3190ac28c8c2271da59106c68 # v6.0.0
        with:
          dotnet-version: ${{ env.DOTNET_VERSION }}

      - name: NuGet audit (G8)
        # Falla el job con Critical/High: el ruleset exige este check, asi que un paquete vulnerable no llega a la rama base.
        shell: bash
        run: |
          dotnet restore "${{ env.SOLUTION_PATH }}"
          dotnet list "${{ env.SOLUTION_PATH }}" package --vulnerable --include-transitive | tee audit.txt
          if grep -qE "Critical|High" audit.txt; then echo "::error::Vulnerabilidades Critical/High en dependencias (G8)"; exit 1; fi
