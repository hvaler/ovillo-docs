# =============================================================================
# .github/actions/security-gate/action.yml — composite action
# G8: descarga el artefacto `security-report` del run y aplica el verdict ANTES del deploy.
#   mode=warn  (dev)     -> vuelca el verdict como warning, NO bloquea.
#   mode=block (pre/pro) -> falla el job si effectiveSummary tiene hallazgos >= gate-severity o secretos > 0.
# Fail-safe: sin verdict valido, en mode=block BLOQUEA (nunca se asume "0 findings").
# Colocar como PRIMER step del job de deploy, tras descargar el artefacto de la app.
# Ovillo cicd-architect github-action-security-gate @ {{TEMPLATE_STAMP}}
# =============================================================================
name: Security gate (Ovillo G8)
description: Lee el verdict-file security-report y bloquea o avisa segun el entorno

inputs:
  mode:
    description: block | warn
    required: false
    default: block
  gate-severity:
    description: severidad efectiva minima que bloquea (Low | Moderate | High | Critical)
    required: false
    default: High
  artifact-name:
    required: false
    default: security-report

runs:
  using: composite
  steps:
    - name: Download security-report
      uses: actions/download-artifact@3e5f45b2cfb9172054b4087a40e8e0b5a5461e7c # v8.0.1
      continue-on-error: true
      with:
        name: ${{ inputs.artifact-name }}
        path: ${{ runner.temp }}/security-report-dl

    - name: Apply verdict
      shell: pwsh
      env:
        MODE: ${{ inputs.mode }}
        GATE: ${{ inputs.gate-severity }}
      run: |
        $report = $null
        $json = Get-ChildItem -Path (Join-Path $env:RUNNER_TEMP 'security-report-dl') -Recurse -Filter 'security-report.json' -ErrorAction SilentlyContinue | Select-Object -First 1
        if ($json) { try { $report = Get-Content $json.FullName -Raw | ConvertFrom-Json } catch { Write-Warning "security-report.json ilegible: $_" } }

        if (-not $report) {
          if ($env:MODE -eq 'block') {
            Write-Host "::error::G8: no hay verdict de seguridad valido. DEPLOY BLOQUEADO (no se asume '0 findings'). Revisar el step Security scan del build."
            exit 1
          }
          Write-Host "::warning::G8: sin verdict de seguridad. mode=warn -> se continua."
          exit 0
        }

        $eff = $report.effectiveSummary
        Write-Host "G8 verdict efectivo: Crit=$($eff.Critical) High=$($eff.High) Mod=$($eff.Moderate) Low=$($eff.Low) Secretos=$($eff.secrets)"
        $order = @{ 'Low' = 1; 'Moderate' = 2; 'High' = 3; 'Critical' = 4 }
        $gateLevel = $order[$env:GATE]
        $blockers = 0
        foreach ($sev in 'Critical','High','Moderate','Low') { if ($order[$sev] -ge $gateLevel) { $blockers += [int]$eff.$sev } }
        $blockers += [int]$eff.secrets
        if ($blockers -gt 0) {
          $msg = "G8: $blockers hallazgo(s) >= $($env:GATE) o secretos en el verdict efectivo."
          if ($env:MODE -eq 'block') {
            Write-Host "::error::$msg DEPLOY BLOQUEADO. Resolver o anadir una excepcion vigente en ops/testing/.security-exceptions.yml."
            exit 1
          }
          Write-Host "::warning::$msg mode=warn -> se despliega igual (dev)."
        } else {
          Write-Host "G8: verdict limpio para gate >= $($env:GATE). Continua el deploy."
        }
