# =============================================================================
# .github/actions/quality-sync/action.yml — composite action
# Publica el Registro de Calidad (QR) del build en el Hub Ovillo (skill calidad-codigo-sync). Best-effort:
# NO tumba el workflow (la calidad no gatea; el gate es el de cobertura). Sin HUB_SERVICE_KEY, avisa y sale 0.
# El script vive en ops/cicd/scripts/Sync-QualityRecord.ps1 (materializado por /cicd-init desde la skill
# calidad-codigo-sync; CI no puede leer el plugin). El dotnet test debe emitir opencover ademas de cobertura
# (ops/testing/coverlet.runsettings lo hace).
# La service-key NO es la apiKey per-dev: es credencial de automatizacion; entra por env, nunca por argumento.
# Ovillo cicd-architect github-action-quality-sync @ {{TEMPLATE_STAMP}}
# =============================================================================
name: Quality record (Ovillo)
description: Publica el Registro de Calidad del build en el Hub Ovillo

inputs:
  project-id:
    description: GUID del proyecto en el Hub (mcpSync.projectId)
    required: true
  hub-service-key:
    description: service-key del Hub (pasar ${{ secrets.HUB_SERVICE_KEY }})
    required: false
    default: ''
  hub-server-url:
    required: false
    default: 'https://hub.example.org'
  results-dir:
    description: carpeta con los *.opencover.xml del dotnet test
    required: false
    default: 'TestResults'
  code-search-base:
    description: base de busqueda de codigo que el dashboard concatena con el fichero
    required: false
    default: ''
  crap-threshold:
    required: false
    default: '30'
  coverage-line-min:
    required: false
    default: '75'
  coverage-branch-min:
    required: false
    default: '60'
  top-hotspots:
    required: false
    default: '20'

runs:
  using: composite
  steps:
    - name: Publish quality record
      shell: pwsh
      continue-on-error: true
      env:
        HUB_SERVICE_KEY: ${{ inputs.hub-service-key }}
        PROJECT_ID: ${{ inputs.project-id }}
        HUB_SERVER_URL: ${{ inputs.hub-server-url }}
        RESULTS_DIR: ${{ inputs.results-dir }}
        CODE_SEARCH_BASE: ${{ inputs.code-search-base }}
        CRAP: ${{ inputs.crap-threshold }}
        COV_LINE: ${{ inputs.coverage-line-min }}
        COV_BRANCH: ${{ inputs.coverage-branch-min }}
        TOP: ${{ inputs.top-hotspots }}
      run: |
        if ([string]::IsNullOrWhiteSpace($env:HUB_SERVICE_KEY)) {
          Write-Host "::warning::HUB_SERVICE_KEY vacia; QR no publicado. Crear el secret: gh secret set HUB_SERVICE_KEY"
          exit 0
        }
        if ([string]::IsNullOrWhiteSpace($env:PROJECT_ID)) {
          Write-Host "::warning::HUB_PROJECT_ID vacio; QR no publicado. Crear la variable: gh variable set HUB_PROJECT_ID --body <GUID>"
          exit 0
        }
        & "$env:GITHUB_WORKSPACE/ops/cicd/scripts/Sync-QualityRecord.ps1" `
          -ProjectId $env:PROJECT_ID `
          -ApiKey $env:HUB_SERVICE_KEY `
          -ResultsDir $env:RESULTS_DIR `
          -CodeSearchBase $env:CODE_SEARCH_BASE `
          -BuildId $env:GITHUB_RUN_NUMBER `
          -ServerUrl $env:HUB_SERVER_URL `
          -CrapThreshold ([int]$env:CRAP) `
          -CoverageLineMin ([int]$env:COV_LINE) `
          -CoverageBranchMin ([int]$env:COV_BRANCH) `
          -TopHotspots ([int]$env:TOP)
