# =============================================================================
# {{PROYECTO_NOMBRE}} — Workflow de seguridad del repositorio (opt-in): CodeQL + dependency review
# Ovillo cicd-architect github-security @ {{TEMPLATE_STAMP}}   (sello de plantilla; /cicd-status detecta drift)
# Reglas: cicd-runtime G15 (permisos minimos, solo el job de CodeQL eleva security-events), G16 (pin por SHA),
# G19 (concurrency). Complementa, no sustituye, el gate G8 (security-scan.ps1 / job Security de ci.yml).
# Materializado por /cicd-init en .github/workflows/security.yml SOLO si el usuario acepta (FASE 2.5-gh, opt-in).
#
# LICENCIA: Code scanning (CodeQL) es gratis en repositorios PUBLICOS; en privados exige GitHub Advanced Security
# (Code Security). Si el repo es privado sin GHAS, el job `codeql` falla en `analyze` al subir el SARIF
# ("Advanced Security must be enabled") -> /cicd-init NO materializa este fichero y deja el gate G8 + Dependabot
# alerts (gratis) como cobertura. `dependency-review` solo necesita el Dependency graph (gratis en todos los planes).
#
# RAMAS CONDICIONALES que /cicd-init aplica:
#   (a) stack Vue o mixto -> anadir 'javascript-typescript' a la matriz de lenguajes.
#   (b) sin PRs (trunk-based sin ruleset) -> quitar el job dependency-review (solo tiene sentido en pull_request).
#   (c) runner Windows self-hosted -> `runs-on: [self-hosted, windows]` y quitar `build-mode: none` si CodeQL no
#       soporta el proyecto sin build (netfx): usar `autobuild`.
# =============================================================================
name: Security

on:
  push:
    branches: ['{{RAMA_BASE}}']
  pull_request:
    branches: ['{{RAMA_BASE}}']
  schedule:
    - cron: '17 4 * * 1'   # lunes 04:17 UTC: detecta CVEs nuevos en dependencias sin cambios de codigo

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  codeql:
    name: CodeQL
    runs-on: '{{RUNNER}}'
    timeout-minutes: 30
    permissions:
      contents: read
      actions: read            # leer el estado de los workflows (lo pide codeql-action)
      security-events: write   # subir el SARIF a Code scanning
      packages: read           # paquetes de CodeQL en GitHub Packages
    strategy:
      fail-fast: false
      matrix:
        include:
          - language: csharp
            build-mode: none   # CodeQL >= 2.16 analiza C# sin compilar; con netfx o generadores de codigo raros usar 'autobuild'
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

      - name: Initialize CodeQL
        uses: github/codeql-action/init@486fec2a3ea2626afcd8c7e9208b4f515078dd7e # codeql-bundle-v2.26.4
        with:
          languages: ${{ matrix.language }}
          build-mode: ${{ matrix.build-mode }}
          queries: security-extended   # 'security-and-quality' si el equipo quiere tambien mantenibilidad

      - name: Analyze
        uses: github/codeql-action/analyze@486fec2a3ea2626afcd8c7e9208b4f515078dd7e # codeql-bundle-v2.26.4
        with:
          category: '/language:${{ matrix.language }}'

  dependency-review:
    name: Dependency review
    if: github.event_name == 'pull_request'
    runs-on: '{{RUNNER}}'
    timeout-minutes: 10
    permissions:
      contents: read
      pull-requests: write     # comentario resumen en la PR
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

      - name: 'Dependency review (G8: bloquea dependencias nuevas con CVE >= high o licencia no permitida)'
        uses: actions/dependency-review-action@a1d282b36b6f3519aa1f3fc636f609c47dddb294 # v5.0.0
        with:
          fail-on-severity: high
          comment-summary-in-pr: on-failure
          # deny-licenses: GPL-3.0, AGPL-3.0   # politica de licencias de la organizacion, si la hay
