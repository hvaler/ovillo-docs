#!/usr/bin/env node
// diagnostico.js - Mide un repositorio ANTES de adoptar Ovillo. No escribe nada.
//
// Existe porque lo caro de adoptar no es arreglar, es ENTERARSE. En un proyecto real la memoria
// siempre cargada eran 125.853 tokens -- nueve veces el techo -- y no habia forma de verlo desde
// dentro: `measure-context.js` vive en el repositorio del constructor y no viaja.
//
// Lo que busca son defectos que NO DAN SINTOMA:
//   - `globs:` en el frontmatter es INERTE (la clave que acota es `paths:`), asi que la unidad
//     carga en todas las sesiones creyendo estar acotada.
//   - `paths: []` tampoco acota: array vacio = siempre.
//   - un BOM antes del `---` puede impedir que se lea el frontmatter entero.
//   - un `@import` crece sin tope y se carga ENTERO en cada sesion.
// Ninguno de los cuatro falla, ni avisa, ni se nota. Solo cuestan contexto, cada vez.
//
// Uso:  node <ruta>/diagnostico.js [--json] [--root <dir>]
// Exit: 0 siempre. Esto informa; decidir es de quien lee (y arreglar, de quien decide).

'use strict';
const fs = require('fs');
const path = require('path');

const ARGS = process.argv.slice(2);
const JSON_OUT = ARGS.includes('--json');
const ROOT = (() => {
  const i = ARGS.indexOf('--root');
  if (i >= 0 && ARGS[i + 1]) return path.resolve(ARGS[i + 1]);
  return process.env.CLAUDE_PROJECT_DIR || process.cwd();
})();

// El catalogo vigente esta donde vive este script: <raiz>/skills/adoptar/scripts/ -> <raiz>/skills.
// Asi vale igual con el plugin y con el zip, sin depender de CLAUDE_PLUGIN_ROOT, que en la shell
// de un script NO existe (ADR-063).
const CATALOGO = path.resolve(__dirname, '..', '..');

const CH_POR_TOKEN = 4;
const toks = (ch) => Math.round(ch / CH_POR_TOKEN);
const TECHO_TOK = 14000;          // el techo que Ovillo se impone al contexto siempre cargado
const IMPORT_MAX_CH = 8000;       // lo que ADR-095 fija para un @import que puede crecer

function leer(p) {
  try { return fs.readFileSync(p, 'utf8'); } catch (e) { return null; }
}
function tieneBom(p) {
  try {
    const fd = fs.openSync(p, 'r');
    const b = Buffer.alloc(3);
    fs.readSync(fd, b, 0, 3, 0);
    fs.closeSync(fd);
    return b[0] === 0xEF && b[1] === 0xBB && b[2] === 0xBF;
  } catch (e) { return false; }
}
function md(dir) {
  const out = [];
  const anda = (d) => {
    let ents;
    try { ents = fs.readdirSync(d, { withFileTypes: true }); } catch (e) { return; }
    for (const e of ents.sort((a, b) => (a.name.toLowerCase() < b.name.toLowerCase() ? -1 : 1))) {
      const p = path.join(d, e.name);
      if (e.isDirectory()) anda(p);
      else if (e.name.toLowerCase().endsWith('.md')) out.push(p);
    }
  };
  anda(dir);
  return out;
}
function frontmatter(texto) {
  const m = /^---\s*\n([\s\S]*?)\n---/.exec(texto.replace(/^﻿/, ''));
  return m ? m[1] : null;
}

const R = { root: ROOT, contexto: {}, inertes: [], boms: [], imports: [], previo: {}, layout: {} };

// ------------------------------------------------------------ A. contexto siempre cargado
(() => {
  const claude = path.join(ROOT, 'CLAUDE.md');
  const texto = leer(claude);
  if (texto === null) { R.contexto.sinClaudeMd = true; return; }

  const piezas = [{ ruta: 'CLAUDE.md', ch: Buffer.byteLength(texto, 'utf8') }];
  for (const linea of texto.split('\n')) {
    const m = /^@(\S+)/.exec(linea.trim());
    if (!m) continue;
    const rel = m[1];
    const abs = path.join(ROOT, rel);
    const ch = fs.existsSync(abs) ? fs.statSync(abs).size : null;
    piezas.push({ ruta: rel, ch, esImport: true });
    R.imports.push({ ruta: rel, ch, excede: ch !== null && ch > IMPORT_MAX_CH });
  }
  R.contexto.piezas = piezas;
  R.contexto.totalCh = piezas.reduce((a, p) => a + (p.ch || 0), 0);
  R.contexto.totalTok = toks(R.contexto.totalCh);
})();

// ------------------------------------- B y C. claves inertes y BOM, en lo que define contexto
(() => {
  // Solo las UNIDADES, no lo que llevan dentro: `skills/<n>/SKILL.md` y los `.md` sueltos de
  // `rules/` y `agents/`. Un `references/x.md` o un `patterns/y.md` es un recurso empaquetado:
  // no tiene frontmatter porque no le toca, y no entra en contexto por su cuenta. La primera
  // version recorria el arbol entero y daba 100+ falsos positivos que enterraban los de verdad --
  // que es el aviso inatendible de ADR-069 fabricado por mi.
  const unidades = [];
  const skillsDir = path.join(ROOT, '.claude', 'skills');
  if (fs.existsSync(skillsDir)) {
    for (const e of fs.readdirSync(skillsDir, { withFileTypes: true })) {
      if (!e.isDirectory()) continue;
      const f = path.join(skillsDir, e.name, 'SKILL.md');
      if (fs.existsSync(f)) unidades.push(f);
    }
  }
  for (const d of ['rules', 'agents']) {
    const dir = path.join(ROOT, '.claude', d);
    if (!fs.existsSync(dir)) continue;
    for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
      if (e.isFile() && e.name.toLowerCase().endsWith('.md')) unidades.push(path.join(dir, e.name));
    }
  }

  let mirados = 0;
  {
    for (const f of unidades) {
      const texto = leer(f);
      if (texto === null) continue;
      mirados++;
      const rel = path.relative(ROOT, f).split(path.sep).join('/');
      if (tieneBom(f)) R.boms.push(rel);
      const fm = frontmatter(texto);
      if (fm === null) {
        if (/^#/.test(texto.replace(/^﻿/, ''))) R.inertes.push({ ruta: rel, motivo: 'sin frontmatter: nada la acota' });
        continue;
      }
      if (/^globs:/m.test(fm)) {
        R.inertes.push({ ruta: rel, motivo: '`globs:` es inerte; la clave que acota es `paths:`' });
      }
      const p = /^paths:\s*(\[\s*\])?\s*$/m.exec(fm);
      if (p && p[1]) R.inertes.push({ ruta: rel, motivo: '`paths: []` no acota: array vacio = carga siempre' });
    }
  }
  // Un cero solo vale si pudo ser otra cosa: hay que decir cuantas se miraron (ADR-044, ADR-078).
  R.inertesMirados = mirados;
})();

// ------------------------------------------------------- E. ecosistema previo, y su mapeo
(() => {
  const dir = (d) => {
    const p = path.join(ROOT, '.claude', d);
    if (!fs.existsSync(p)) return [];
    try {
      return fs.readdirSync(p, { withFileTypes: true })
        .filter((e) => (d === 'skills' ? e.isDirectory() : e.isFile()))
        .map((e) => e.name.replace(/\.(md|js|ps1)$/i, ''));
    } catch (e) { return []; }
  };
  const previo = { commands: dir('commands'), skills: dir('skills'), hooks: dir('hooks'), rules: dir('rules'), agents: dir('agents') };

  let catalogo = null;
  try {
    catalogo = new Set(fs.readdirSync(CATALOGO, { withFileTypes: true })
      .filter((e) => e.isDirectory() && fs.existsSync(path.join(CATALOGO, e.name, 'SKILL.md')))
      .map((e) => e.name));
  } catch (e) { catalogo = null; }

  previo.total = previo.commands.length + previo.skills.length + previo.hooks.length
    + previo.rules.length + previo.agents.length;

  // Los hooks se mapean contra los hooks, no contra las skills. La primera version solo miraba
  // comandos y skills, y en el caso real dejaba fuera 16 hooks de los que mapeaban LOS 16.
  let catHooks = null;
  try {
    catHooks = new Set(fs.readdirSync(path.join(CATALOGO, '..', 'hooks'), { withFileTypes: true })
      .filter((e) => e.isFile() && e.name.endsWith('.js'))
      .map((e) => e.name.replace(/\.js$/, '')));
  } catch (e) { catHooks = null; }

  if (catHooks && catHooks.size && previo.hooks.length) {
    previo.hooksMapean = previo.hooks.filter((n) => catHooks.has(n));
    previo.hooksSinMapear = previo.hooks.filter((n) => !catHooks.has(n));
    previo.catalogoHooks = catHooks.size;
  }

  if (catalogo && catalogo.size) {
    const candidatos = previo.commands.concat(previo.skills);
    previo.mapean = candidatos.filter((n) => catalogo.has(n));
    previo.sinMapear = candidatos.filter((n) => !catalogo.has(n));
    previo.catalogo = catalogo.size;
  } else {
    // Decirlo, en vez de publicar un cero que parece "no hay nada que mapear".
    previo.mapeoImposible = 'no se encontro el catalogo de skills junto a este script';
  }
  R.previo = previo;
})();

// --------------------------------------------------------------------- F. layout
(() => {
  const esperado = ['src', 'docs', 'ops', '_hilo', '_patron'];
  R.layout.presentes = esperado.filter((d) => fs.existsSync(path.join(ROOT, d)));
  R.layout.faltan = esperado.filter((d) => !fs.existsSync(path.join(ROOT, d)));

  // Un Hilo con otro nombre sigue siendo un Hilo. Sin esto, un proyecto que YA tiene memoria de
  // proyecto --bajo `_duran`, `_memoria`, lo que sea-- se lee como si no tuviera ninguna, y la
  // adopcion se plantea como "montar" cuando en realidad es "renombrar y actualizar". Fue el
  // hallazgo mas util del caso real y la primera version no lo veia.
  const CONTRATO = ['ESTADO_PROYECTO.json', 'SESION_ACTUAL.md', 'DECISIONES.md', 'LECCIONES.md',
    'HISTORIAL_CAMBIOS.md', 'FUNCIONALIDADES.md', 'DEPENDENCIAS.md'];
  R.layout.hilosAlternativos = [];
  try {
    for (const e of fs.readdirSync(ROOT, { withFileTypes: true })) {
      if (!e.isDirectory() || e.name === '_hilo' || e.name.startsWith('.')) continue;
      let hay = 0;
      for (const f of CONTRATO) if (fs.existsSync(path.join(ROOT, e.name, f))) hay++;
      if (hay >= 3) R.layout.hilosAlternativos.push({ dir: e.name, coinciden: hay, de: CONTRATO.length });
    }
  } catch (e) { /* raiz ilegible */ }
  // Donde esta el codigo de verdad: el .csproj mas somero que no sea de test.
  const proyectos = [];
  const anda = (d, prof) => {
    if (prof > 4) return;
    let ents;
    try { ents = fs.readdirSync(d, { withFileTypes: true }); } catch (e) { return; }
    for (const e of ents) {
      if (e.name === 'node_modules' || e.name === 'bin' || e.name === 'obj' || e.name.startsWith('.')) continue;
      const p = path.join(d, e.name);
      if (e.isDirectory()) anda(p, prof + 1);
      else if (/\.csproj$/i.test(e.name) && !/tests?\./i.test(e.name)) {
        proyectos.push(path.relative(ROOT, p).split(path.sep).join('/'));
      }
    }
  };
  anda(ROOT, 0);
  R.layout.proyectos = proyectos.slice(0, 8);
  R.layout.codigoEnRaizSrc = proyectos.some((p) => p.startsWith('src/'));
})();

// --------------------------------------------------------------------- informe
if (JSON_OUT) {
  console.log(JSON.stringify(R, null, 2));
  process.exit(0);
}

const L = [];
L.push('Diagnostico de adopcion  ·  ' + ROOT);
L.push('');

L.push('CONTEXTO SIEMPRE CARGADO');
if (R.contexto.sinClaudeMd) {
  L.push('  no hay CLAUDE.md: el proyecto parte de cero y no arrastra contexto.');
} else {
  for (const p of R.contexto.piezas) {
    const marca = p.ch === null ? '  (NO EXISTE)' : '';
    L.push('  ' + (p.esImport ? '@' : ' ') + ' ' + p.ruta.padEnd(42)
      + String(p.ch === null ? '-' : p.ch).padStart(8) + ' ch  ~'
      + String(p.ch === null ? '-' : toks(p.ch)).padStart(6) + ' tok' + marca);
  }
  const t = R.contexto.totalTok;
  L.push('  ' + ''.padEnd(44) + String(R.contexto.totalCh).padStart(8) + ' ch  ~' + String(t).padStart(6) + ' tok'
    + (t > TECHO_TOK ? '   PASA del techo de ' + TECHO_TOK + ' en ' + (t - TECHO_TOK) : '   (techo ' + TECHO_TOK + ')'));
  const gordos = R.imports.filter((i) => i.excede);
  if (gordos.length) {
    L.push('');
    L.push('  Un @import se carga ENTERO en cada sesion. Estos pasan de ' + IMPORT_MAX_CH + ' ch:');
    for (const i of gordos) L.push('    - ' + i.ruta + '  ' + i.ch + ' ch (~' + toks(i.ch) + ' tok)');
  }
}
L.push('');

L.push('CLAVES QUE NO ACOTAN  (miradas: ' + R.inertesMirados + ')');
if (!R.inertesMirados) L.push('  no hay skills, reglas ni agents propios que mirar.');
else if (!R.inertes.length) L.push('  ninguna: las ' + R.inertesMirados + ' miradas acotan o cargan siempre a proposito.');
else for (const i of R.inertes) L.push('  - ' + i.ruta.padEnd(44) + '  ' + i.motivo);
L.push('');

L.push('BOM ANTES DEL FRONTMATTER');
if (!R.boms.length) L.push('  ninguno.');
else {
  L.push('  ' + R.boms.length + ' fichero(s); un BOM antes del `---` puede impedir que se lea el frontmatter entero:');
  for (const b of R.boms.slice(0, 10)) L.push('    - ' + b);
}
L.push('');

L.push('ECOSISTEMA PREVIO');
if (!R.previo.total) L.push('  ninguno: no hay .claude/ con comandos, skills, hooks, reglas ni agents.');
else {
  for (const k of ['commands', 'skills', 'hooks', 'rules', 'agents']) {
    if (R.previo[k].length) L.push('  ' + k.padEnd(10) + String(R.previo[k].length).padStart(3));
  }
  if (R.previo.mapeoImposible) {
    L.push('  ' + R.previo.mapeoImposible + ': el mapeo NO se ha hecho (no es que no haya).');
  } else {
    L.push('  de ' + (R.previo.commands.length + R.previo.skills.length) + ' comandos/skills, '
      + R.previo.mapean.length + ' coinciden por nombre con el catalogo de ' + R.previo.catalogo + '.');
    if (R.previo.sinMapear.length) {
      L.push('  sin coincidencia por nombre (muchos existen renombrados; hay que mirarlos):');
      L.push('    ' + R.previo.sinMapear.join(', '));
    }
  }
  if (R.previo.hooksMapean) {
    L.push('  de ' + R.previo.hooks.length + ' hooks, ' + R.previo.hooksMapean.length
      + ' coinciden por nombre con los ' + R.previo.catalogoHooks + ' del ecosistema.');
    if (R.previo.hooksSinMapear.length) {
      L.push('    sin coincidencia: ' + R.previo.hooksSinMapear.join(', '));
    }
  }
}
L.push('');

L.push('LAYOUT');
L.push('  presentes: ' + (R.layout.presentes.join(' ') || '(ninguno)'));
L.push('  faltan   : ' + (R.layout.faltan.join(' ') || '(ninguno)'));
for (const h of R.layout.hilosAlternativos || []) {
  L.push('  `' + h.dir + '/` tiene ' + h.coinciden + ' de los ' + h.de + ' ficheros del contrato del Hilo:');
  L.push('    es un Hilo con otro nombre. Adoptar aqui es RENOMBRAR Y ACTUALIZAR, no montar de cero.');
}
if (R.layout.proyectos.length) {
  L.push('  codigo   : ' + R.layout.proyectos.slice(0, 4).join(', ')
    + (R.layout.proyectos.length > 4 ? ' ...' : ''));
  if (!R.layout.codigoEnRaizSrc) {
    L.push('  el codigo NO esta bajo src/ en la raiz. Ovillo lo espera ahi y no es configurable');
    L.push('  (ADR-005). Moverlo es una decision aparte: este informe no mueve nada.');
  }
}
L.push('');
L.push('Esto no ha escrito nada. Avisar y arreglar son cosas distintas.');

console.log(L.join('\n'));
process.exit(0);
