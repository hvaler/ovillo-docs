// ============================================================================
// TEMPLATE: ArchitectureTests.cs - las reglas de Vertical Slices, ejecutables
// Skill: testing-patterns (ADR-046)
// ============================================================================
// USO: Reemplazar {{ProjectNamespace}}. Se genera UNA VEZ por proyecto.
// Destino: src/tests/{{ProjectNamespace}}.Tests.Architecture/ArchitectureTests.cs
// Aplica cuando `configuracion.arquitectura.estilo` del Hilo vale `vertical-slices`.
//
// ESTE JUEGO NO ES UNA VARIANTE DEL DE CLEAN: ES SU CONTRARIO.
//
// En Clean las reglas defienden las capas horizontales: Domain no conoce Infrastructure, el
// contrato del repositorio vive aqui y no alla. En Vertical Slices esas capas NO EXISTEN a
// proposito, y aplicarles las reglas de Clean seria peor que no tener ninguna: rojo permanente
// sobre un proyecto perfectamente sano.
//
// Lo que este estilo defiende es otra cosa: que una feature no conozca a otra. El acoplamiento
// DENTRO de una slice es barato y deseable -- para cambiar "crear pedido" abres un fichero, no
// cuatro carpetas --; entre slices es lo que convierte el proyecto en un monolito enredado sin que
// nadie tome la decision de enredarlo.
//
// PAQUETE:
//   <PackageReference Include="NetArchTest.eNhancedEdition" Version="1.4.5" />
// NO `NetArchTest.Rules` (sin publicar desde 2021 y sin licencia en el nuspec). Y aqui importa mas
// que en Clean: la primitiva `Slice()` que hace la regla principal es del fork, el original no la
// tiene.
//
// EL FALLO CLASICO, y como se evita: si un filtro no selecciona ningun tipo, la regla se cumple
// sobre el conjunto vacio y el test pasa EN VERDE sin vigilar nada. `Cumple()` exige por eso que la
// regla haya mirado al menos un tipo. La regla de slices lo comprueba aparte, contando features:
// con una sola slice "ninguna feature depende de otra" es cierto y no significa nada.

using System.Reflection;
using NetArchTest.Rules;
using Xunit;

namespace {{ProjectNamespace}}.Tests.Architecture;

public class ArchitectureTests
{
    private const string Features = "{{ProjectNamespace}}.Features";
    private const string Comun = "{{ProjectNamespace}}.Common";
    private const string Persistencia = "{{ProjectNamespace}}.Persistence";

    private static readonly Assembly[] Ensamblados = new[]
    {
        typeof({{ProjectNamespace}}.Common.ErrorKind).Assembly,
        // Common es lo unico que TODAS las slices referencian y que existe en los dos modos de
        // acceso a datos, asi que es el ancla estable. Antes habia una segunda a
        // `Persistence.AppDbContext`, y con procedimientos ese tipo no existe: las reglas, que
        // hablan de independencia entre features, no compilaban sin un ORM detras (ADR-054).
        // Si tu proyecto parte Persistence en su propio ensamblado, anade aqui un tipo suyo.
    }.Distinct().ToArray();

    private static Types Todos() => Types.InAssemblies(Ensamblados);

    /// <summary>Afirma la regla Y que la regla ha mirado algo.</summary>
    private static void Cumple(TestResult resultado, string regla)
    {
        var mirados = resultado.SelectedTypesForTesting?.Count() ?? 0;
        Assert.True(mirados > 0,
            $"{regla}\n  La regla no selecciono ningun tipo: el filtro esta mal o la capa no existe. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        if (resultado.IsSuccessful) return;

        var culpables = string.Join("\n    ", resultado.FailingTypes.Select(t => t.FullName));
        Assert.True(false, $"{regla}\n  Lo incumplen:\n    {culpables}");
    }

    /// <summary>
    /// La afirmacion contraria: que un namespace este VACIO. `Cumple()` no sirve para esto, porque
    /// su guardia contra el verde vacio exige justo lo contrario -- que haya tipos seleccionados.
    /// </summary>
    private static void NoExisteNada(string espacio, string regla)
    {
        var intrusos = Todos().That().ResideInNamespace(espacio).GetTypes()
            .Select(t => t.FullName).OrderBy(n => n).ToList();
        Assert.True(intrusos.Count == 0,
            $"{regla}\n  Existen {intrusos.Count} tipo(s) ahi:\n    " + string.Join("\n    ", intrusos));
    }

    // ---------------------------------------------------------------- la regla que define el estilo

    [Fact]
    public void Ninguna_feature_conoce_a_otra()
    {
        // Contar las slices primero. Con NINGUNA, el filtro esta mal y hay que decirlo: ese es el
        // verde vacio de verdad. Con UNA, "ninguna depende de otra" es trivialmente cierto y el test
        // no afirma nada -- pero eso no es un defecto, es un proyecto de una sola feature, que es
        // exactamente lo que genera un CRUD de una entidad. Fallar ahi ponia en rojo el producto.
        var slices = Todos().That().ResideInNamespace(Features).GetTypes()
            .Select(t => t.ReflectionType.Namespace)   // IType no expone Namespace; ReflectionType es el Type real
            .Where(n => n is not null && n.Length > Features.Length + 1)
            .Select(n => n!.Substring(Features.Length + 1).Split('.')[0])
            .Distinct().OrderBy(n => n).ToList();

        Assert.True(slices.Count >= 1,
            $"No hay ninguna feature bajo {Features}: el filtro esta mal o el prefijo no es ese. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        var r = Todos().Slice().ByNamespacePrefix(Features)
            .Should().NotHaveDependenciesBetweenSlices()
            .GetResult();

        Assert.True(r.IsSuccessful,
            "Una feature conoce a otra, y eso es lo que este estilo existe para impedir.\n" +
            "  Si dos features necesitan lo mismo, sube a Common POR DECISION, o comunicalas por un\n" +
            "  identificador o un evento. Una referencia directa de tipos convierte dos slices en una.\n" +
            "  Features: " + string.Join(", ", slices));
    }

    // ---------------------------------------------------------------- lo compartido no mira hacia abajo

    [Fact]
    public void Lo_compartido_no_conoce_ninguna_feature()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Comun)
            .ShouldNot().HaveDependencyOnAny(Features)
            .GetResult(),
            "En cuanto Common conoce una feature deja de ser un nucleo compartido y pasa a ser el " +
            "concentrador que acopla todo con todo por la puerta de atras.");
    }

    [Fact]
    public void La_persistencia_no_conoce_ninguna_feature()
    {
        // DOS FORMAS VALIDAS, y por eso esta regla no puede exigir que el namespace exista:
        //
        //   con ORM  -> hay un `Persistence` compartido (contexto y mapeos), y entonces la regla
        //               muerde: ese almacen sirve a las features y no sabe cuales hay.
        //   con SP   -> NO hay capa de persistencia. Cada slice abre su conexion e invoca sus
        //               procedimientos, que es lo maximo que este estilo puede prometer.
        //
        // Exigir el namespace pondria en rojo una configuracion correcta del estilo; darlo por bueno
        // sin mirar dejaria pasar el caso real. Se comprueba cual de las dos es (ADR-054).
        var enPersistencia = Todos().That().ResideInNamespace(Persistencia).GetTypes().Any();
        if (!enPersistencia) return;

        Cumple(Todos()
            .That().ResideInNamespace(Persistencia)
            .ShouldNot().HaveDependencyOnAny(Features)
            .GetResult(),
            "El almacen sirve a las features; no sabe cuales hay.");
    }

    // ---------------------------------------------------------------- que no vuelvan las capas

    [Fact]
    public void No_hay_capas_horizontales()
    {
        // La deriva tipica: alguien echa de menos la estructura conocida y empieza a crear
        // Application/ o Infrastructure/ al lado de Features/. A partir de ahi el proyecto no es ni
        // una cosa ni la otra. Si de verdad hacen falta esas capas, la decision es cambiar de estilo
        // en el Hilo -- y entonces aplica el otro juego de reglas --, no mezclarlas en silencio.
        //
        // DOMAIN NO ESTA EN LA LISTA, y es deliberado. Un modelo de dominio compartido no es una capa
        // horizontal colandose: es lo que HACE POSIBLE la independencia entre slices. Si cada feature
        // tuviera su propia entidad, la que necesitara la de otra tendria que referenciar
        // Features.Otra -- y eso incumple la regla principal. El estilo corta el COMPORTAMIENTO, no
        // el modelo. Prohibir Domain aqui ponia en rojo el CRUD que el propio ecosistema genera.
        foreach (var capa in new[] { "Application", "Infrastructure" })
        {
            NoExisteNada($"{{ProjectNamespace}}.{capa}",
                $"En Vertical Slices no hay capa {capa}: lo que la feature necesita vive con la feature.");
        }
    }

    [Fact]
    public void Todo_tipo_de_Features_pertenece_a_una_feature()
    {
        var sueltos = Todos().That().ResideInNamespace(Features).GetTypes()
            .Where(t => t.ReflectionType.Namespace == Features)
            .Select(t => t.FullName).OrderBy(n => n).ToList();

        Assert.True(sueltos.Count == 0,
            "Hay tipos colgando de la raiz de Features, sin pertenecer a ninguna slice. O son de una " +
            "feature concreta y bajan a ella, o son compartidos y suben a Common:\n    " +
            string.Join("\n    ", sueltos));
    }

    // ---------------------------------------------------------------- forma de los tipos

    [Fact]
    public void Los_handlers_son_sealed()
    {
        Cumple(Todos()
            .That().AreClasses().And().HaveNameEndingWith("Handler")
            .Should().BeSealed()
            .GetResult(),
            "Un handler es el final de una cadena, no una base de la que heredar.");
    }
}

// ============================================================================
// LO QUE NO ESTA AQUI, Y POR QUE
//
// No hay regla sobre donde vive el contrato HTTP. En Vertical Slices el request del caso de uso ES
// a menudo el cuerpo de la peticion, y esa fusion es una decision del estilo, no un descuido: se
// paga versionando la slice entera cuando cambia la API. Si tu proyecto quiere separarlos, es una
// regla que puedes anadir; no se impone porque no es lo que el estilo dice.
//
// Tampoco hay reglas de direccion entre capas, porque no hay capas. Ese es el juego de `clean`.
// ============================================================================
