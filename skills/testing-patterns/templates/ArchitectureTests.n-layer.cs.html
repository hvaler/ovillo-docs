// ============================================================================
// TEMPLATE: ArchitectureTests.cs - las reglas de N capas, ejecutables
// Skill: testing-patterns (ADR-048)
// ============================================================================
// USO: Reemplazar {{ProjectNamespace}}. Se genera UNA VEZ por proyecto.
// Destino: src/tests/{{ProjectNamespace}}.Tests.Architecture/ArchitectureTests.cs
// Aplica cuando `configuracion.arquitectura.estilo` del Hilo vale `n-layer`.
//
// SON CUATRO REGLAS, Y ESO NO ES UN DESCUIDO.
//
// N capas tiene menos que vigilar que Clean porque promete menos. No hay inversion de dependencias,
// no hay puertos, no hay contratos repartidos: hay tres capas y las dependencias van de arriba
// abajo. Inflar este juego para que tuviera trece reglas seria fingir una cobertura que el estilo
// no pide.
//
// LA TRAMPA, y es la que hay que evitar al leer esto con ojos de Clean:
//
//   En n-layer, QUE LA CAPA DE NEGOCIO DEPENDA DE LA DE DATOS ES CORRECTO. Es el estilo
//   funcionando, no un defecto. La propia guia de Microsoft lo dice al presentarlo: la desventaja
//   de este modelo es justamente que las dependencias de compilacion van de arriba abajo, con la
//   logica de negocio dependiendo de detalles de acceso a datos. Quien no quiera eso no quiere
//   n capas: quiere Clean, y entonces cambia el estilo en el Hilo y le aplica el otro juego.
//
// Lo que si se vigila es lo que rompe el modelo en sus propios terminos: SALTARSE una capa y MIRAR
// hacia arriba.
//
// PAQUETE:
//   <PackageReference Include="NetArchTest.eNhancedEdition" Version="1.4.5" />
// NO `NetArchTest.Rules` (sin publicar desde 2021 y sin licencia en el nuspec).

using System.Reflection;
using NetArchTest.Rules;
using Xunit;

namespace {{ProjectNamespace}}.Tests.Architecture;

public class ArchitectureTests
{
    // Los namespaces que emite `generador-crud` para este estilo, alineados con el perfil BASICO del
    // wizard (`Api -> Services -> Repository -> BD`, dos proyectos). Si tu proyecto usa los nombres
    // tradicionales UI / BLL / DAL, cambia estas tres constantes y las reglas valen igual.
    private const string Presentacion = "{{ProjectNamespace}}.Api";
    private const string Negocio = "{{ProjectNamespace}}.Core.Services";
    private const string Datos = "{{ProjectNamespace}}.Core.Data";

    private static readonly Assembly[] Ensamblados = new[]
    {
        // Ancla en la capa de negocio, NO en el contexto de EF Core: este estilo se genera tambien
        // con Dapper y procedimientos (ADR-052), donde no hay `AppDbContext` y estas reglas -- que
        // hablan de capas, no de tecnologia -- no compilarian siquiera. Sustituye `Scholarship` por
        // cualquier entidad tuya: solo sirve para localizar el ensamblado.
        typeof({{ProjectNamespace}}.Core.Services.I{{Entity}}Service).Assembly,
    }.Distinct().ToArray();

    private static Types Todos() => Types.InAssemblies(Ensamblados);

    /// <summary>Afirma la regla Y que la regla ha mirado algo.</summary>
    private static void Cumple(TestResult resultado, string regla)
    {
        var mirados = resultado.SelectedTypesForTesting?.Count() ?? 0;
        Assert.True(mirados > 0,
            $"{regla}\n  La regla no selecciono ningun tipo: el filtro esta mal o la capa no existe. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        if (resultado.IsSuccessful) return;

        var culpables = string.Join("\n    ", resultado.FailingTypes.Select(t => t.FullName));
        Assert.True(false, $"{regla}\n  Lo incumplen:\n    {culpables}");
    }

    // ---------------------------------------------------------------- no saltarse capas

    [Fact]
    public void La_presentacion_no_salta_a_la_capa_de_datos()
    {
        // La regla del estilo. Saltarse la capa de negocio para ir directo al repositorio es lo que
        // convierte una arquitectura de tres capas en tres carpetas: a partir de ahi la logica se
        // reparte entre la UI y el servicio, y nadie sabe donde esta la de verdad.
        Cumple(Todos()
            .That().ResideInNamespace(Presentacion)
            .ShouldNot().HaveDependencyOnAny(Datos)
            .GetResult(),
            "La presentacion habla con la capa de negocio, y solo con ella.");
    }

    [Fact]
    public void La_presentacion_no_toca_el_acceso_a_datos_directamente()
    {
        // El corolario concreto de la anterior, y el que se incumple de verdad: un DbContext
        // inyectado en el controlador "solo para esta consulta rapida".
        Cumple(Todos()
            .That().ResideInNamespace(Presentacion)
            .ShouldNot().HaveDependencyOnAny(
                "Microsoft.EntityFrameworkCore", "System.Data", "Microsoft.Data.SqlClient")
            .GetResult(),
            "Una consulta rapida desde el controlador es como empieza siempre; luego son veinte.");
    }

    // ---------------------------------------------------------------- no mirar hacia arriba

    [Fact]
    public void La_capa_de_datos_no_mira_hacia_arriba()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Datos)
            .ShouldNot().HaveDependencyOnAny(Negocio, Presentacion)
            .GetResult(),
            "La capa de abajo nunca sabe quien la llama: si lo sabe, deja de ser reutilizable y " +
            "cualquier cambio de arriba la arrastra.");
    }

    [Fact]
    public void La_capa_de_negocio_no_conoce_la_presentacion()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Negocio)
            .ShouldNot().HaveDependencyOnAny(Presentacion)
            .GetResult(),
            "Un servicio que devuelve un ActionResult o lee el HttpContext ya no es logica de " +
            "negocio: es un controlador con otro nombre.");
    }
}

// ============================================================================
// LO QUE NO ESTA AQUI, Y POR QUE
//
// No hay ninguna regla que prohiba a la capa de negocio depender de la de datos. En n capas eso es
// lo correcto, y una regla asi pondria en rojo un proyecto sano. Si lo que se quiere es invertir esa
// dependencia, el estilo que se busca es `clean`: se cambia en el Hilo y aplica su juego.
//
// Tampoco hay reglas de nomenclatura de interfaces ni de forma de los tipos. N capas no promete
// nada sobre eso, y una regla que el estilo no pide acaba en una lista de exclusiones.
// ============================================================================
