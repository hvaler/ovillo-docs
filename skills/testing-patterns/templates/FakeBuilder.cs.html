// Generic builder with Bogus for test data
// Placeholders: {{EntityName}}, {{ProjectNamespace}}

using Bogus;

namespace {{ProjectNamespace}}.Tests.Builders;

public class FakeBuilder<T> where T : class
{
    protected readonly Faker<T> Faker;

    public FakeBuilder(Faker<T> faker)
    {
        Faker = faker;
        Faker.UseSeed(42);
    }

    public T Build() => Faker.Generate();
    public List<T> BuildMany(int count = 5) => Faker.Generate(count);
}
