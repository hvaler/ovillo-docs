// ============================================================================
// TEMPLATE: ArchitectureTests.cs - las reglas de capa, ejecutables
// Skill: testing-patterns (ADR-044)
// ============================================================================
// USO: Reemplazar {{ProjectNamespace}}. Se genera UNA VEZ por proyecto.
// Destino: src/tests/{{ProjectNamespace}}.Tests.Architecture/ArchitectureTests.cs
//
// POR QUE EXISTE:
//
// Las reglas de capa del ecosistema estaban escritas en prosa (el checklist de
// analisis-arquitectura) y las revisaba un agente. Una regla que solo revisa un humano o un
// agente se viola el dia que nadie mira. Estos tests las ponen donde no se pueden ignorar:
// en `dotnet test`, o sea en el pipeline, en cada PR.
//
// PAQUETE, y por que este:
//
//   <PackageReference Include="NetArchTest.eNhancedEdition" Version="1.4.5" />
//
// NO `NetArchTest.Rules`. Ese es el original y su ultima version es de 2021-05-23, con el
// nuspec SIN licencia declarada. `eNhancedEdition` es su fork, MIT, con publicaciones
// recientes, y conserva el namespace `NetArchTest.Rules`: los `using` no cambian.
//
// EL FALLO CLASICO DE ESTOS TESTS, y como se evita aqui:
//
// Si el filtro no selecciona ningun tipo -un namespace mal escrito, un renombrado, una capa
// que aun no existe- la regla se cumple sobre el conjunto vacio y el test pasa EN VERDE.
// A partir de ahi no vigila nada y nadie se entera. Por eso `Cumple()` exige ademas que la
// regla haya mirado al menos un tipo: un test de arquitectura que no selecciona nada esta
// roto, no conforme.

using System.Reflection;
using NetArchTest.Rules;
using Xunit;

namespace {{ProjectNamespace}}.Tests.Architecture;

public class ArchitectureTests
{
    private const string Dominio = "{{ProjectNamespace}}.Domain";
    private const string Aplicacion = "{{ProjectNamespace}}.Application";
    private const string Infraestructura = "{{ProjectNamespace}}.Infrastructure";
    private const string Api = "{{ProjectNamespace}}.Api";

    // Un tipo ancla por capa, solo para localizar los ensamblados. Con cuatro proyectos son
    // cuatro; con uno solo, `Distinct()` lo deja en uno. Las reglas filtran por NAMESPACE, asi
    // que valen igual en los dos montajes y siguen valiendo si el proyecto se parte mas tarde.
    private static readonly Assembly[] Ensamblados = new[]
    {
        typeof({{ProjectNamespace}}.Domain.Entities.AuditableEntity).Assembly,
        typeof({{ProjectNamespace}}.Application.Common.Interfaces.IUnitOfWork).Assembly,
        // Ancla en el repositorio y NO en el contexto de EF Core: este estilo se genera tambien con
        // Dapper y procedimientos (ADR-053), donde no hay `ApplicationDbContext` y estas reglas -- que
        // hablan de capas, no de tecnologia -- no compilarian siquiera. Sustituye `Scholarship` por
        // cualquier entidad tuya: solo sirve para localizar el ensamblado.
        typeof({{ProjectNamespace}}.Infrastructure.Persistence.Repositories.{{Entity}}Repository).Assembly,
        typeof({{ProjectNamespace}}.Api.Filters.LoggingFilter).Assembly,
    }.Distinct().ToArray();

    private static Types Todos() => Types.InAssemblies(Ensamblados);

    /// <summary>
    /// Afirma la regla Y que la regla ha mirado algo. Sin la segunda mitad, un filtro que no
    /// casa con nada produce un test verde que no vigila nada.
    /// </summary>
    private static void Cumple(TestResult resultado, string regla)
    {
        var mirados = resultado.SelectedTypesForTesting?.Count() ?? 0;
        Assert.True(mirados > 0,
            $"{regla}\n  La regla no selecciono ningun tipo: el filtro esta mal o la capa no existe. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        if (resultado.IsSuccessful) return;

        var culpables = string.Join("\n    ", resultado.FailingTypes.Select(t => t.FullName));
        Assert.True(false, $"{regla}\n  Lo incumplen:\n    {culpables}");
    }

    // ---------------------------------------------------------------- direccion de dependencias

    [Fact]
    public void El_dominio_no_conoce_ninguna_otra_capa()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Dominio)
            .ShouldNot().HaveDependencyOnAny(Aplicacion, Infraestructura, Api)
            .GetResult(),
            "Domain es el centro: las dependencias entran, no salen.");
    }

    [Fact]
    public void El_dominio_no_conoce_EF_Core()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Dominio)
            .ShouldNot().HaveDependencyOnAny("Microsoft.EntityFrameworkCore")
            .GetResult(),
            "El dominio se persiste, no sabe persistirse. EF Core vive en Infrastructure.");
    }

    [Fact]
    public void El_dominio_no_conoce_HTTP_ni_serializacion()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Dominio)
            .ShouldNot().HaveDependencyOnAny(
                "Microsoft.AspNetCore", "System.Text.Json", "Newtonsoft.Json")
            .GetResult(),
            "Una entidad con atributos de serializacion deja de ser del dominio y pasa a ser del transporte.");
    }

    [Fact]
    public void La_aplicacion_no_conoce_ni_Infrastructure_ni_Api()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Aplicacion)
            .ShouldNot().HaveDependencyOnAny(Infraestructura, Api)
            .GetResult(),
            "Los casos de uso hablan con interfaces; quien las implementa es cosa del arranque.");
    }

    [Fact]
    public void La_aplicacion_no_conoce_EF_Core()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Aplicacion)
            .ShouldNot().HaveDependencyOnAny("Microsoft.EntityFrameworkCore")
            .GetResult(),
            "Un IQueryable que se escapa de Infrastructure convierte el repositorio en decorado. " +
            "Si este test se pone rojo, la consulta se ha ido de capa.");
    }

    [Fact]
    public void Infrastructure_no_conoce_la_Api()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Infraestructura)
            .ShouldNot().HaveDependencyOnAny(Api)
            .GetResult(),
            "Infrastructure sirve a Application, no al transporte que hoy tenga delante.");
    }

    // ---------------------------------------------------------------- donde vive cada cosa

    [Fact]
    public void Las_implementaciones_de_repositorio_viven_en_Infrastructure()
    {
        Cumple(Todos()
            .That().AreClasses().And().HaveNameEndingWith("Repository")
            .Should().ResideInNamespace(Infraestructura)
            .GetResult(),
            "Una clase Repository fuera de Infrastructure es acceso a datos filtrado a otra capa.");
    }

    [Fact]
    public void Los_contratos_de_repositorio_viven_en_el_dominio()
    {
        // El contrato de un repositorio de agregado es lenguaje del negocio ("dame el pedido",
        // "guarda el pedido"), asi que pertenece a la capa que define ese lenguaje. Los puertos
        // TECNICOS del caso de uso -- IUnitOfWork, ICurrentUserService, IClock -- se quedan en
        // Application, que es de quien son. Si un contrato de repositorio acaba en Infrastructure
        // junto a su implementacion, la inversion de dependencias deja de invertir nada.
        Cumple(Todos()
            .That().AreInterfaces().And().HaveNameEndingWith("Repository")
            .Should().ResideInNamespace(Dominio + ".Repositories")
            .GetResult(),
            "Un repositorio de agregado se declara donde vive su lenguaje: el dominio.");
    }

    [Fact]
    public void Las_lecturas_paginadas_son_contrato_de_Application()
    {
        // La contrapartida de la regla anterior: una lectura paginada devuelve un modelo de lectura,
        // no un agregado, y es del caso de uso. Tenerla en el repositorio de dominio obligaria al
        // dominio a conocer los tipos de pagina, que viven en Application.
        Cumple(Todos()
            .That().AreInterfaces().And().HaveNameEndingWith("Queries")
            .Should().ResideInNamespace(Aplicacion)
            .GetResult(),
            "Las consultas de pantalla son del caso de uso, no del dominio.");
    }

    [Fact]
    public void Los_controllers_solo_viven_en_la_Api()
    {
        Cumple(Todos()
            .That().AreClasses().And().HaveNameEndingWith("Controller")
            .Should().ResideInNamespace(Api)
            .GetResult(),
            "Un Controller fuera de Api es transporte metido donde no toca.");
    }

    // ---------------------------------------------------------------- forma de los tipos

    [Fact]
    public void Las_entidades_del_dominio_heredan_de_AuditableEntity()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Dominio + ".Entities")
                .And().AreClasses().And().AreNotAbstract()
            .Should().Inherit(typeof({{ProjectNamespace}}.Domain.Entities.AuditableEntity))
            .GetResult(),
            "Quien y cuando toco una fila no es opcional: sale de la base, no de acordarse.");
    }

    [Fact]
    public void Los_handlers_son_sealed()
    {
        Cumple(Todos()
            .That().AreClasses().And().HaveNameEndingWith("Handler")
            .Should().BeSealed()
            .GetResult(),
            "Un handler es el final de una cadena, no una base de la que heredar.");
    }

    [Fact]
    public void Las_interfaces_empiezan_por_I()
    {
        Cumple(Todos()
            .That().AreInterfaces().And().ResideInNamespace("{{ProjectNamespace}}")
            .Should().HaveNameStartingWith("I")
            .GetResult(),
            "Convencion de nomenclatura de la organizacion.");
    }
}

// ============================================================================
// COMO ANADIR UNA REGLA
//
// Una regla nueva solo merece la pena si es plausible que alguien la incumpla por descuido.
// Antes de dejarla fija, comprobar que FALLA: introducir la violacion a proposito, ver el
// test rojo, y deshacerla. Un test de arquitectura que nunca ha estado rojo no se ha probado.
//
// Reglas que NO van aqui porque NetArchTest no las ve: llamadas concretas dentro de un metodo
// (`DateTime.Now`, `Thread.Sleep`, `.Result`), que son de un analizador de Roslyn; y el
// contenido de las cadenas SQL, que es del hook `sql-nomenclatura-guard`.
// ============================================================================
