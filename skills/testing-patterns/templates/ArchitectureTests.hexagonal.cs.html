// ============================================================================
// TEMPLATE: ArchitectureTests.cs - las reglas de Puertos y Adaptadores, ejecutables
// Skill: testing-patterns (ADR-047)
// ============================================================================
// USO: Reemplazar {{ProjectNamespace}}. Se genera UNA VEZ por proyecto.
// Destino: src/tests/{{ProjectNamespace}}.Tests.Architecture/ArchitectureTests.cs
// Aplica cuando `configuracion.arquitectura.estilo` del Hilo vale `hexagonal`.
//
// POR QUE ESTE JUEGO NO ES EL DE CLEAN CON OTROS NOMBRES:
//
// Es la suposicion que costo una version. Clean y Hexagonal comparten la direccion -- las
// dependencias apuntan al centro -- pero no comparten la regla que las distingue, y tampoco los
// namespaces: aqui no hay Domain/Application/Infrastructure/Api, hay un NUCLEO con sus PUERTOS y
// unos ADAPTADORES a su alrededor.
//
// La regla propia es esta: NINGUN ADAPTADOR CONOCE A OTRO. En Clean se tolera -- y la propia guia de
// Microsoft lo dice -- que la capa de UI referencie Infrastructure en el composition root. Aqui eso
// es justo lo que el estilo existe para impedir: cada adaptador se enchufa al hexagono, no a otro
// adaptador. El dia que el controlador llama al repositorio saltandose el caso de uso, el hexagono
// deja de serlo aunque los `using` sigan apuntando hacia dentro.
//
// LA ASIMETRIA DE LOS PUERTOS, que es lo que hay que entender para leer estas reglas:
//   de ENTRADA (driving): lo que el exterior puede pedir. Lo IMPLEMENTA el nucleo.
//   de SALIDA  (driven):  lo que la aplicacion necesita. Lo DECLARA el nucleo, lo implementa fuera.
// En los dos casos la interfaz es del nucleo. Por eso se cambia la API o la base de datos sin tocar
// el caso de uso.
//
// PAQUETE:
//   <PackageReference Include="NetArchTest.eNhancedEdition" Version="1.4.5" />
// NO `NetArchTest.Rules` (sin publicar desde 2021 y sin licencia en el nuspec); ademas la primitiva
// `Slice()` que hace la regla principal es del fork.

using System.Reflection;
using NetArchTest.Rules;
using Xunit;

namespace {{ProjectNamespace}}.Tests.Architecture;

public class ArchitectureTests
{
    private const string Nucleo = "{{ProjectNamespace}}.Core";
    private const string Dominio = "{{ProjectNamespace}}.Core.Domain";
    private const string Puertos = "{{ProjectNamespace}}.Core.Ports";
    private const string CasosDeUso = "{{ProjectNamespace}}.Core.UseCases";
    private const string Adaptadores = "{{ProjectNamespace}}.Adapters";
    private const string Salida = "{{ProjectNamespace}}.Adapters.Out";

    private static readonly Assembly[] Ensamblados = new[]
    {
        typeof({{ProjectNamespace}}.Core.Domain.DomainException).Assembly,
    }.Distinct().ToArray();

    private static Types Todos() => Types.InAssemblies(Ensamblados);

    /// <summary>Afirma la regla Y que la regla ha mirado algo.</summary>
    private static void Cumple(TestResult resultado, string regla)
    {
        var mirados = resultado.SelectedTypesForTesting?.Count() ?? 0;
        Assert.True(mirados > 0,
            $"{regla}\n  La regla no selecciono ningun tipo: el filtro esta mal o esa parte no existe. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        if (resultado.IsSuccessful) return;

        var culpables = string.Join("\n    ", resultado.FailingTypes.Select(t => t.FullName));
        Assert.True(false, $"{regla}\n  Lo incumplen:\n    {culpables}");
    }

    /// <summary>Los grupos que hay bajo un prefijo, para no afirmar nada sobre un solo grupo.</summary>
    private static List<string> GruposBajo(string prefijo) =>
        Todos().That().ResideInNamespace(prefijo).GetTypes()
            .Select(t => t.ReflectionType.Namespace)   // IType no expone Namespace; ReflectionType es el Type real
            .Where(n => n is not null && n.Length > prefijo.Length + 1)
            .Select(n => n!.Substring(prefijo.Length + 1).Split('.')[0])
            .Distinct().OrderBy(n => n).ToList();

    // ---------------------------------------------------------------- la regla que define el estilo

    [Fact]
    public void Ningun_adaptador_conoce_a_otro()
    {
        // (a) Entrada frente a salida: que el controlador no llame al repositorio por la espalda.
        var lados = GruposBajo(Adaptadores);
        Assert.True(lados.Count >= 2,
            $"Solo hay un lado de adaptadores ({string.Join(", ", lados)}). Se esperan al menos " +
            $"{Adaptadores}.In y {Adaptadores}.Out; con uno solo la regla no afirma nada.");

        Assert.True(Todos().Slice().ByNamespacePrefix(Adaptadores)
            .Should().NotHaveDependenciesBetweenSlices().GetResult().IsSuccessful,
            "Un adaptador conoce a otro, y eso es lo que este estilo existe para impedir.\n" +
            "  El caso tipico: el adaptador de entrada llama al de salida saltandose el caso de uso.\n" +
            "  A partir de ahi el hexagono deja de serlo aunque los using apunten hacia dentro.\n" +
            "  Lados: " + string.Join(", ", lados));

        // (b) Y entre los de salida: dos adaptadores driven tampoco se conocen entre si.
        var deSalida = GruposBajo(Salida);
        if (deSalida.Count >= 2)
        {
            Assert.True(Todos().Slice().ByNamespacePrefix(Salida)
                .Should().NotHaveDependenciesBetweenSlices().GetResult().IsSuccessful,
                "Dos adaptadores de salida se conocen entre si. Si uno necesita al otro, la " +
                "composicion es cosa del nucleo o del arranque, no de los adaptadores.\n" +
                "  De salida: " + string.Join(", ", deSalida));
        }
    }

    // ---------------------------------------------------------------- la direccion, hacia dentro

    [Fact]
    public void El_nucleo_no_conoce_ningun_adaptador()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Nucleo)
            .ShouldNot().HaveDependencyOnAny(Adaptadores)
            .GetResult(),
            "El nucleo declara lo que necesita y espera que alguien lo enchufe; no sabe quien.");
    }

    [Fact]
    public void El_nucleo_no_conoce_HTTP_ni_persistencia_ni_serializacion()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Nucleo)
            .ShouldNot().HaveDependencyOnAny(
                "Microsoft.AspNetCore", "Microsoft.EntityFrameworkCore",
                "System.Text.Json", "Newtonsoft.Json", "System.Data")
            .GetResult(),
            "Si el nucleo conoce la tecnologia, el puerto ya no aisla de nada.");
    }

    [Fact]
    public void El_dominio_no_conoce_los_casos_de_uso()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Dominio)
            .ShouldNot().HaveDependencyOnAny(CasosDeUso, Puertos)
            .GetResult(),
            "El dominio es el centro del centro: las reglas de negocio no saben que casos de uso " +
            "las invocan, ni que puertos hacen falta para servirlos.");
    }

    // ---------------------------------------------------------------- los puertos, donde tocan

    [Fact]
    public void Todo_puerto_es_una_interfaz()
    {
        Cumple(Todos()
            .That().ResideInNamespace(Puertos)
            .Should().BeInterfaces()
            .GetResult(),
            "Un puerto es un contrato. Una clase ahi dentro es una implementacion colandose al nucleo.");
    }

    [Fact]
    public void Todo_puerto_vive_en_el_nucleo()
    {
        Cumple(Todos()
            .That().AreInterfaces().And().HaveNameEndingWith("Port")
            .Should().ResideInNamespace(Puertos)
            .GetResult(),
            "Un puerto declarado fuera del nucleo es un adaptador dictandole al nucleo lo que necesita, " +
            "que es la dependencia al reves.");
    }
}

// ============================================================================
// LO QUE NO ESTA AQUI, Y POR QUE
//
// No hay regla que obligue a cada adaptador de salida a implementar un puerto. Se puede expresar,
// pero casa mal: un adaptador legitimo puede tener tipos auxiliares que no implementan nada, y la
// regla daria rojos falsos que acabarian en exclusiones. La direccion ya la vigilan las de arriba.
//
// Tampoco hay reglas de capas horizontales: no las hay. Ese es el juego de `clean`.
// ============================================================================
