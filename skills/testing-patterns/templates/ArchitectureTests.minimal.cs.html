// ============================================================================
// TEMPLATE: ArchitectureTests.cs - las dos reglas de una app minima, ejecutables
// Skill: testing-patterns (ADR-048)
// ============================================================================
// USO: Reemplazar {{ProjectNamespace}}. Se genera UNA VEZ por proyecto.
// Destino: src/tests/{{ProjectNamespace}}.Tests.Architecture/ArchitectureTests.cs
// Aplica cuando `configuracion.arquitectura.estilo` del Hilo vale `minimal`.
//
// SON DOS REGLAS. NO FALTAN LAS DEMAS: ES QUE NO LAS HAY.
//
// Este es el juego mas corto de todos, y la brevedad es el contenido. Un proyecto unico no tiene
// fronteras de ensamblado que vigilar; toda la separacion es de carpetas, y eso son convenciones,
// no invariantes. Escribir aqui trece reglas para que se parezca a `clean` seria fingir una
// cobertura que el estilo no pide y que ademas estorbaria.
//
// LA REGLA QUE NO ESTA, Y QUE CASI ESCRIBO:
//
//   "Los endpoints no tocan el acceso a datos directamente."
//
// Es FALSA aqui. En una app de un proyecto, un `MapGet` que consulta el contexto y devuelve es el
// patron idiomatico y recomendado; ponerle un servicio en medio "porque toca" es la ceremonia que
// este estilo existe para evitar. Una regla asi pondria en rojo un proyecto perfectamente sano.
//
// Es el mismo error que habria sido aplicarle a `n-layer` la inversion de dependencias de Clean, o a
// `hexagonal` los nombres de las cuatro capas. Cada estilo se vigila con SUS reglas o con ninguna.
//
// LO QUE SI QUEDA son dos cosas, y son las de abajo: una direccion que sigue importando aunque no
// haya capas, y el aviso de que el proyecto ha dejado de ser minimal.
//
// PAQUETE:
//   <PackageReference Include="NetArchTest.eNhancedEdition" Version="1.4.5" />

using System.Reflection;
using NetArchTest.Rules;
using Xunit;

namespace {{ProjectNamespace}}.Tests.Architecture;

public class ArchitectureTests
{
    private const string Endpoints = "{{ProjectNamespace}}.Endpoints";
    private const string Datos = "{{ProjectNamespace}}.Data";

    private static readonly Assembly[] Ensamblados = new[]
    {
        // Ancla en la ENTIDAD y no en el contexto de EF Core: este estilo se genera tambien con
        // Dapper y procedimientos (ADR-054), donde no hay `AppDbContext` -- y estas reglas, que
        // hablan de la forma del proyecto y no de la tecnologia, no compilarian siquiera.
        // Sustituye `Scholarship` por cualquier entidad tuya: solo sirve para localizar el ensamblado.
        typeof({{ProjectNamespace}}.Data.{{Entity}}).Assembly,
    }.Distinct().ToArray();

    private static Types Todos() => Types.InAssemblies(Ensamblados);

    /// <summary>Afirma la regla Y que la regla ha mirado algo.</summary>
    private static void Cumple(TestResult resultado, string regla)
    {
        var mirados = resultado.SelectedTypesForTesting?.Count() ?? 0;
        Assert.True(mirados > 0,
            $"{regla}\n  La regla no selecciono ningun tipo: el filtro esta mal o esa parte no existe. " +
            "Un test de arquitectura que no mira nada esta roto, no conforme.");

        if (resultado.IsSuccessful) return;

        var culpables = string.Join("\n    ", resultado.FailingTypes.Select(t => t.FullName));
        Assert.True(false, $"{regla}\n  Lo incumplen:\n    {culpables}");
    }

    // ---------------------------------------------------------------- la direccion que queda

    [Fact]
    public void El_acceso_a_datos_no_conoce_los_endpoints()
    {
        // Al reves si: el endpoint habla con el almacen directamente y esta bien. Pero el almacen
        // mirando al endpoint es el unico acoplamiento que sigue siendo un error aunque no haya
        // capas, porque es el que impide sacar el acceso a datos el dia que el proyecto crezca.
        Cumple(Todos()
            .That().ResideInNamespace(Datos)
            .ShouldNot().HaveDependencyOnAny(Endpoints)
            .GetResult(),
            "El almacen no sabe quien lo consulta. Es lo unico que hay que mantener si algun dia " +
            "esto deja de ser un solo proyecto.");
    }

    // ---------------------------------------------------------------- el aviso de que ya no es minimal

    [Fact]
    public void El_proyecto_sigue_siendo_minimal()
    {
        // No vigila una violacion: vigila que la ETIQUETA siga siendo cierta. El dia que aparecen
        // capas, el proyecto ha crecido y lo honesto es cambiar el estilo en el Hilo -- y recibir
        // entonces el juego de reglas que si le corresponde -- en vez de quedarse con `minimal`
        // declarado y una arquitectura que ya es otra cosa.
        var capas = new[] { "Application", "Domain", "Infrastructure", "Business" };
        var encontradas = new List<string>();
        foreach (var capa in capas)
        {
            var tipos = Todos().That().ResideInNamespace($"{{ProjectNamespace}}.{capa}").GetTypes().ToList();
            if (tipos.Count > 0) encontradas.Add($"{capa} ({tipos.Count} tipos)");
        }

        Assert.True(encontradas.Count == 0,
            "Han aparecido capas: " + string.Join(", ", encontradas) + ".\n" +
            "  Esto ya no es una app minima, y el estilo declarado en el Hilo ha dejado de describirla.\n" +
            "  Cambiar `configuracion.arquitectura.estilo` a `n-layer` o `clean` segun hacia donde haya\n" +
            "  crecido: el juego de reglas correspondiente si tiene algo que decir sobre esas capas.\n" +
            "  Este test no se arregla borrando las capas, sino decidiendo cual es ahora el estilo.");
    }
}

// ============================================================================
// CUANDO ESTE FICHERO SE QUEDA CORTO
//
// Si echas de menos reglas, probablemente el proyecto ya no sea minimal. Eso no es un fallo de la
// plantilla: es la senal para la que existe la segunda regla. Cambia el estilo en el Hilo y
// `ArchitectureTests.<estilo>.cs.template` te dara el juego que corresponda.
// ============================================================================
