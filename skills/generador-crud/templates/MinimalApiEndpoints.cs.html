// TEMPLATE: MinimalApiEndpoints.cs - Endpoints Minimal API con TypedResults (por entidad; alternativa: Controller.cs)
// Skill: generador-crud (ADR-031)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{entities}}, {{Namespace}}
// Destino: src/Api/Endpoints/{{Entities}}Endpoints.cs
// Misma convencion de versionado que el controlador: /api/v{version}/{{entities}} (Asp.Versioning.Http).
// ============================================================================

using Asp.Versioning;
using Asp.Versioning.Builder;
using Mediator;
using Microsoft.AspNetCore.Http.HttpResults;
using {{Namespace}}.Api.Filters;
using {{Namespace}}.Application.Common.Models;
using {{Namespace}}.Application.Features.{{Entities}}.Commands;
using {{Namespace}}.Application.Features.{{Entities}}.Dtos;
using {{Namespace}}.Application.Features.{{Entities}}.Queries;

namespace {{Namespace}}.Api.Endpoints;

public static class {{Entities}}Endpoints
{
    public static IEndpointRouteBuilder Map{{Entities}}Endpoints(this IEndpointRouteBuilder app)
    {
        var api = app.NewVersionedApi("{{Entities}}");
        var group = api.MapGroup("/api/v{version:apiVersion}/{{entities}}")
            .HasApiVersion(new ApiVersion(1, 0))
            .WithTags("{{Entities}}")
            .RequireAuthorization()
            .WithLogging();

        group.MapGet("/", GetAll).WithName("Get{{Entities}}");
        group.MapGet("/{id:int}", GetById).WithName("Get{{Entity}}ById");
        group.MapPost("/", Create).WithName("Create{{Entity}}").WithValidation<Create{{Entity}}Command>();
        group.MapPut("/{id:int}", Update).WithName("Update{{Entity}}").WithValidation<Update{{Entity}}Command>();
        group.MapDelete("/{id:int}", Delete).WithName("Delete{{Entity}}");
        return app;
    }

    private static async Task<Ok<PaginatedResult<{{Entity}}Dto>>> GetAll(
        [AsParameters] Get{{Entities}}Query query,
        IMediator mediator, CancellationToken ct)
    {
        var result = await mediator.Send(query, ct);
        return TypedResults.Ok(result);
    }

    private static async Task<Results<Ok<{{Entity}}Dto>, NotFound>> GetById(
        int id, IMediator mediator, CancellationToken ct)
    {
        var result = await mediator.Send(new Get{{Entity}}ByIdQuery(id), ct);
        return result is not null ? TypedResults.Ok(result) : TypedResults.NotFound();
    }

    private static async Task<Created<{{Entity}}Dto>> Create(
        Create{{Entity}}Command command,
        IMediator mediator, CancellationToken ct)
    {
        var result = await mediator.Send(command, ct);
        return TypedResults.Created($"/api/v1/{{entities}}/{result.Id}", result);
    }

    private static async Task<Results<NoContent, BadRequest<string>>> Update(
        int id, Update{{Entity}}Command command,
        IMediator mediator, CancellationToken ct)
    {
        if (command.Id != 0 && command.Id != id)
        {
            return TypedResults.BadRequest("The id in the route does not match the id in the body");
        }
        await mediator.Send(command with { Id = id }, ct);
        return TypedResults.NoContent();
    }

    private static async Task<NoContent> Delete(
        int id, IMediator mediator, CancellationToken ct)
    {
        await mediator.Send(new Delete{{Entity}}Command(id), ct);
        return TypedResults.NoContent();
    }
}
