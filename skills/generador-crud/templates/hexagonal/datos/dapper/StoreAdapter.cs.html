// ============================================================================
// TEMPLATE: {{Entity}}StoreAdapter.cs - adaptador de salida con Dapper y SQL propio (por entidad)
// Skill: generador-crud, estilo hexagonal con acceso por SQL (ADR-055)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{Namespace}}, {{Schema}} y los fragmentos
//      {{ColumnasSp}}, {{ValoresSpCrear}}, {{ParametrosSpCrear_Valores}}, {{AsignacionesSpActualizar}},
//      {{ParametrosDapperCrear}}, {{ParametrosDapperActualizar}}
// Destino: src/Adapters/Out/Persistence/{{Entity}}StoreAdapter.cs
//
// EL MISMO PUERTO, LA TERCERA IMPLEMENTACION: `Ef{{Entity}}StoreAdapter` (EF Core),
// `Sp{{Entity}}StoreAdapter` (procedimientos) y este. El nucleo (`Core/`) no se toca en ninguna de las
// tres, que es lo unico que el hexagono promete y aqui se cobra.
//
// DIFERENCIA CON EL MODO `dapper-sp`, y es la unica: **donde vive el SQL**. Alli lo guarda la base y
// el C# invoca un nombre; aqui lo guarda este fichero. Lo demas -- conexion por operacion,
// materializacion, pagina y total en una ida -- es identico.
//
// EL TOPE DE PAGINA SE APLICA AQUI Y BASTA. Con procedimientos habia que ponerlo tambien en el
// procedimiento porque cualquiera podia invocarlo sin pasar por la aplicacion; una cadena SQL dentro
// del binario no es superficie publica, asi que acotar en un sitio acota de verdad.
//
// PARAMETRIZADO SIEMPRE, sin excepcion: los valores viajan como parametros de Dapper (`@Name`), nunca
// interpolados en la cadena. Concatenar aqui es inyeccion SQL, y da igual que el valor «venga de
// dentro»: el dia que venga de fuera nadie se acordara de esta linea.
// ============================================================================

using System.Data;
using Dapper;
using Microsoft.Data.SqlClient;
using {{Namespace}}.Core.Domain;
using {{Namespace}}.Core.Ports;

namespace {{Namespace}}.Adapters.Out.Persistence;

public sealed class Sql{{Entity}}StoreAdapter : I{{Entity}}StorePort
{
    /// <summary>Tope de pagina: sin el, un `size=100000` convierte el listado en un incidente.</summary>
    private const int MaxPorPagina = 200;

    // ─────────────────────────────────────────────────────────────────────── el SQL, en un solo sitio
    // Agrupado arriba y no repartido por los metodos: asi se lee de un vistazo que toca esta clase, y
    // un DBA puede revisarlo sin leer C#.

    private const string SqlGet = """
        SELECT {{ColumnasSp}}
        FROM {{Schema}}.{{Entities}}
        WHERE Id = @Id;
        """;

    // El desempate por Id evita que una fila salte de pagina cuando dos comparten CreatedAt.
    private const string SqlPagina = """
        SELECT {{ColumnasSp}}
        FROM {{Schema}}.{{Entities}}
        ORDER BY CreatedAt DESC, Id ASC
        OFFSET (@Page - 1) * @Size ROWS
        FETCH NEXT @Size ROWS ONLY;

        SELECT COUNT(*) FROM {{Schema}}.{{Entities}};
        """;

    private const string SqlCrear = """
        INSERT INTO {{Schema}}.{{Entities}} ({{ValoresSpCrear}}, CreatedAt)
        VALUES ({{ParametrosSpCrear_Valores}}, @CreatedAt);

        SELECT CAST(SCOPE_IDENTITY() AS INT);
        """;

    private const string SqlActualizar = """
        UPDATE {{Schema}}.{{Entities}}
        SET {{AsignacionesSpActualizar}},
            ModifiedAt = @ModifiedAt
        WHERE Id = @Id;

        SELECT @@ROWCOUNT;
        """;

    private const string SqlBorrar = """
        DELETE FROM {{Schema}}.{{Entities}} WHERE Id = @Id;

        SELECT @@ROWCOUNT;
        """;

    private const string SqlExisteNombre = """
        SELECT CASE WHEN EXISTS (
            SELECT 1 FROM {{Schema}}.{{Entities}} WHERE Name = @Name
        ) THEN 1 ELSE 0 END;
        """;

    // ───────────────────────────────────────────────────────────────────────

    private readonly string _cadenaConexion;

    public Sql{{Entity}}StoreAdapter(string cadenaConexion)
    {
        _cadenaConexion = cadenaConexion;
    }

    // Una conexion por operacion. Con el pool de ADO.NET abrir y cerrar no cuesta una conexion real.
    private SqlConnection Conexion() => new(_cadenaConexion);

    public async Task<{{Entity}}?> GetByIdAsync(int id, CancellationToken ct = default)
    {
        using var conexion = Conexion();

        return await conexion.QuerySingleOrDefaultAsync<{{Entity}}>(new CommandDefinition(
            SqlGet, new { Id = id }, cancellationToken: ct));
    }

    public async Task<(IReadOnlyList<{{Entity}}> Items, int Total)> GetPageAsync(int page, int size, CancellationToken ct = default)
    {
        // page = 0 produciria un OFFSET negativo, que SQL Server rechaza.
        page = Math.Max(1, page);
        size = Math.Clamp(size, 1, MaxPorPagina);

        using var conexion = Conexion();

        // Los dos SELECT del mismo lote se leen con QueryMultipleAsync: una ida a la base, no dos.
        using var multiple = await conexion.QueryMultipleAsync(new CommandDefinition(
            SqlPagina, new { Page = page, Size = size }, cancellationToken: ct));

        var filas = (await multiple.ReadAsync<{{Entity}}>()).ToList();
        var total = await multiple.ReadSingleAsync<int>();

        return (filas, total);
    }

    public async Task<{{Entity}}> AddAsync({{Entity}} entity, CancellationToken ct = default)
    {
        // Las marcas de auditoria las pone el DOMINIO en este estilo (la factoria y Update()), no el
        // adaptador ni el motor. Por eso este metodo no las toca (ADR-054 D3).
        using var conexion = Conexion();

        var id = await conexion.ExecuteScalarAsync<int>(new CommandDefinition(
            SqlCrear, {{ParametrosDapperCrear}}, cancellationToken: ct));

        return await GetByIdAsync(id, ct)
            ?? throw new InvalidOperationException(
                $"El INSERT devolvio el id {id} y la lectura posterior no encuentra esa fila.");
    }

    public async Task<bool> UpdateAsync({{Entity}} entity, CancellationToken ct = default)
    {
        using var conexion = Conexion();
        var id = entity.Id;

        var filas = await conexion.ExecuteScalarAsync<int>(new CommandDefinition(
            SqlActualizar, {{ParametrosDapperActualizar}}, cancellationToken: ct));

        return filas > 0;
    }

    public async Task<bool> DeleteAsync(int id, CancellationToken ct = default)
    {
        using var conexion = Conexion();

        var filas = await conexion.ExecuteScalarAsync<int>(new CommandDefinition(
            SqlBorrar, new { Id = id }, cancellationToken: ct));

        return filas > 0;
    }

    public async Task<bool> ExistsNameAsync(string name, CancellationToken ct = default)
    {
        using var conexion = Conexion();

        return await conexion.ExecuteScalarAsync<bool>(new CommandDefinition(
            SqlExisteNombre, new { Name = name }, cancellationToken: ct));
    }
}
