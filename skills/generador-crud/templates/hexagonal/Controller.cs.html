// ============================================================================
// TEMPLATE: {{Entities}}Controller.cs - adaptador de ENTRADA (por entidad)
// Skill: generador-crud, estilo hexagonal (ADR-051)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{entities}}, {{Namespace}}
// Destino: src/Adapters/In/Rest/{{Entities}}Controller.cs
//
// TRADUCE HTTP A UNA LLAMADA AL PUERTO DE ENTRADA, y nada mas. Lo que NO hace, y es la regla que
// define el estilo: hablar con un adaptador de salida. No conoce Ef{{Entity}}Store, ni AppDbContext,
// ni EF Core. En Clean se tolera que la capa de UI referencie Infrastructure en el composition root;
// aqui eso es justo la violacion -- cada adaptador se enchufa al hexagono, no a otro adaptador.
// ============================================================================

using Microsoft.AspNetCore.Mvc;
using {{Namespace}}.Core.Domain;
using {{Namespace}}.Core.Dtos;
using {{Namespace}}.Core.Ports;

namespace {{Namespace}}.Adapters.In.Rest;

[ApiController]
[Route("api/v1/{{entities}}")]
public class {{Entities}}Controller : ControllerBase
{
    private readonly I{{Entity}}UseCasePort _casos;

    public {{Entities}}Controller(I{{Entity}}UseCasePort casos) => _casos = casos;

    [HttpGet("{id:int}")]
    [ProducesResponseType<{{Entity}}Dto>(StatusCodes.Status200OK)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public async Task<ActionResult<{{Entity}}Dto>> GetById(int id, CancellationToken ct = default)
    {
        var dto = await _casos.GetAsync(id, ct);
        return dto is null ? NotFound() : Ok(dto);
    }

    [HttpGet]
    [ProducesResponseType<PagedResult<{{Entity}}Dto>>(StatusCodes.Status200OK)]
    public async Task<ActionResult<PagedResult<{{Entity}}Dto>>> GetPage(
        int page = 1, int size = 20, CancellationToken ct = default)
        => Ok(await _casos.GetPageAsync(page, size, ct));

    [HttpPost]
    [ProducesResponseType<{{Entity}}Dto>(StatusCodes.Status201Created)]
    [ProducesResponseType<ProblemDetails>(StatusCodes.Status409Conflict)]
    public async Task<ActionResult<{{Entity}}Dto>> Create(
        [FromBody] Create{{Entity}}Command request, CancellationToken ct = default)
    {
        try
        {
            var dto = await _casos.CreateAsync(request, ct);
            return CreatedAtAction(nameof(GetById), new { id = dto.Id }, dto);
        }
        catch (DomainException ex)
        {
            // El nucleo senala la regla rota con su propia excepcion; TRADUCIRLA a un codigo HTTP es
            // cosa de este adaptador, no del nucleo. Otro adaptador de entrada la traduciria a otra
            // cosa, y el caso de uso seguiria igual.
            return Conflict(new ProblemDetails
            {
                Title = "Conflict",
                Detail = ex.Message,
                Status = StatusCodes.Status409Conflict,
            });
        }
    }

    [HttpPut("{id:int}")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public async Task<IActionResult> Update(
        int id, [FromBody] Update{{Entity}}Command request, CancellationToken ct = default)
        => await _casos.UpdateAsync(id, request, ct) ? NoContent() : NotFound();

    [HttpDelete("{id:int}")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public async Task<IActionResult> Delete(int id, CancellationToken ct = default)
        => await _casos.DeleteAsync(id, ct) ? NoContent() : NotFound();
}
