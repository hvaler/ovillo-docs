// TEMPLATE: {{Entity}}Repository.cs - capa de datos con ADO.NET puro y procedimientos (por entidad)
// Skill: generador-crud, acceso a datos por ADO.NET (ADR-056)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{Namespace}}, {{Schema}} y los fragmentos
//      {{LecturaReader}}, {{ParametrosAdoCrear}}, {{ParametrosAdoActualizar}}
// Destino: src/Core/Data/{{Entity}}Repository.cs
//
// La interfaz es la MISMA que implementan el repositorio de EF Core, el de Dapper y el de Dapper con
// procedimientos. El servicio no cambia ni una linea, y eso es lo que la capa de datos existe para dar.
//
// ESTE MODO ES EL MAS LARGO A PROPOSITO, y la guia de acceso a datos de la organizacion ya lo dice:
// «codigo boilerplate: mucho». Si puedes usar Dapper, usa Dapper: hace exactamente esto.
//
// LO QUE ESTE MODO EVITA ES UN PAQUETE DE TERCEROS, no las dependencias. Sigue necesitando
// `Microsoft.Data.SqlClient`, que es mayor que Dapper: lo unico que se ahorra es Dapper. La
// restriccion a la que sirve es «nada fuera de Microsoft», que en algunas casas es una politica
// real, o un codigo heredado que ya es ADO.NET y al que el codigo nuevo tiene que parecerse. Son
// dos de los cuatro casos que la guia de acceso a datos de la organizacion recoge.
//
// NO ES POR RENDIMIENTO: esa misma guia mide la diferencia con Dapper en menos del 5 %. Si la
// razon para elegir este modo fuera la velocidad, la razon seria mala.
//
// LOS PARAMETROS VAN TIPADOS, nunca con `AddWithValue`. `AddWithValue` infiere el tipo del valor de
// C#, y una `string` se manda como `nvarchar(4000)`: si la columna es `varchar(200)`, SQL Server hace
// una conversion implicita, el indice deja de servir y la consulta pasa de buscar a escanear. Es el
// error clasico de ADO.NET y no da ningun sintoma hasta que la tabla crece.
//
// LAS COLUMNAS SE LEEN POR NOMBRE (`GetOrdinal`), no por posicion. Con posiciones, anadir una columna
// al procedimiento desplaza todas las demas y el codigo sigue compilando: empieza a leer la fecha
// donde estaba el nombre. El coste de `GetOrdinal` es una busqueda en un diccionario por fila.
// ============================================================================

        cmd.CommandType = CommandType.StoredProcedure;

    public async Task<{{Entity}}?> GetByIdAsync(int id, CancellationToken ct = default)

        var (cn, cmd) = await AbrirAsync("{{Schema}}.Get{{Entity}}", ct);
        await using (cn)
        await using (cmd)

    public async Task<(IReadOnlyList<{{Entity}}> Items, int Total)> GetPageAsync(
        int page, int size, CancellationToken ct = default)
    {
        page = Math.Max(1, page);
        size = Math.Clamp(size, 1, MaxPorPagina);

        var (cn, cmd) = await AbrirAsync("{{Schema}}.List{{Entities}}", ct);
        await using (cn)
        await using (cmd)
        {
            cmd.Parameters.Add("@Page", SqlDbType.Int).Value = page;
            cmd.Parameters.Add("@Size", SqlDbType.Int).Value = size;

            await using var reader = await cmd.ExecuteReaderAsync(ct);

            // El procedimiento devuelve DOS conjuntos -- la pagina y el total -- y se recorren con
            // NextResultAsync. Con dos llamadas separadas el total podria no corresponder a la pagina.
            var items = new List<{{Entity}}>();
            while (await reader.ReadAsync(ct)) items.Add(Leer(reader));

            var total = 0;
            if (await reader.NextResultAsync(ct) && await reader.ReadAsync(ct))
            {
                total = reader.GetInt32(0);
            }

            return (items, total);
        }
    }
