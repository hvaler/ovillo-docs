// ============================================================================
// TEMPLATE: Controller.cs - Controlador API REST (por entidad; alternativa: MinimalApiEndpoints.cs)
// Skill: generador-crud (ADR-031)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{Namespace}}
// Destino: src/Api/Controllers/{{Entities}}Controller.cs
// ============================================================================

using Asp.Versioning;
using Mediator;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using {{Namespace}}.Api.Contracts.{{Entities}};
using {{Namespace}}.Application.Common.Models;
using {{Namespace}}.Application.Features.{{Entities}}.Commands;
using {{Namespace}}.Application.Features.{{Entities}}.Dtos;
using {{Namespace}}.Application.Features.{{Entities}}.Queries;

namespace {{Namespace}}.Api.Controllers;

/// <summary>
/// API REST para gestión de {{Entities}}
/// </summary>
[ApiController]
[ApiVersion("1.0")]
[Route("api/v{version:apiVersion}/[controller]")]
[Produces("application/json")]
public class {{Entities}}Controller : ControllerBase
{
    private readonly IMediator _mediator;
    private readonly ILogger<{{Entities}}Controller> _logger;

    private readonly {{Entity}}ContractMapper _contratos;

    public {{Entities}}Controller(
        IMediator mediator,
        ILogger<{{Entities}}Controller> logger,
        {{Entity}}ContractMapper contratos)
    {
        _mediator = mediator;
        _logger = logger;
        _contratos = contratos;
    }

    // ═══════════════════════════════════════════════════════════════════════
    // GET - Consultas
    // ═══════════════════════════════════════════════════════════════════════

    /// <summary>
    /// Obtiene listado paginado de {{Entities}}
    /// </summary>
    /// <param name="page">Número de página (1-based)</param>
    /// <param name="size">Elementos por página</param>
    /// <param name="ct">Token de cancelación</param>
    [HttpGet]
    [ProducesResponseType<PaginatedResult<{{Entity}}Dto>>(StatusCodes.Status200OK)]
    public async Task<ActionResult<PaginatedResult<{{Entity}}Dto>>> GetAll(
        [FromQuery] int page = 1,
        [FromQuery] int size = 10,
        CancellationToken ct = default)
    {
        var result = await _mediator.Send(new Get{{Entities}}Query(page, size), ct);
        return Ok(result);
    }

    /// <summary>
    /// Obtiene un/a {{Entity}} por su ID
    /// </summary>
    [HttpGet("{id:int}")]
    [ProducesResponseType<{{Entity}}Dto>(StatusCodes.Status200OK)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public async Task<ActionResult<{{Entity}}Dto>> GetById(
        int id,
        CancellationToken ct = default)
    {
        var result = await _mediator.Send(new Get{{Entity}}ByIdQuery(id), ct);

        if (result is null)
        {
            return NotFound(new ProblemDetails
            {
                Title = "{{Entity}} not found",
                Detail = $"No {{Entity}} with id {id}",
                Status = StatusCodes.Status404NotFound
            });
        }

        return Ok(result);
    }

    // ═══════════════════════════════════════════════════════════════════════
    // POST - Crear
    // ═══════════════════════════════════════════════════════════════════════

    /// <summary>
    /// Crea un/a nuevo/a {{Entity}}
    /// </summary>
    [HttpPost]
    [Authorize(Roles = "Admin,Manager")]
    [ProducesResponseType<{{Entity}}Dto>(StatusCodes.Status201Created)]
    [ProducesResponseType<ValidationProblemDetails>(StatusCodes.Status400BadRequest)]
    public async Task<ActionResult<{{Entity}}Dto>> Create(
        [FromBody] Create{{Entity}}Request request,
        CancellationToken ct = default)
    {
        var result = await _mediator.Send(_contratos.ToCommand(request), ct);

        _logger.LogInformation("{{Entity}} {Id} created by {User}", result.Id, User.Identity?.Name);

        return CreatedAtAction(nameof(GetById), new { id = result.Id }, result);
    }

    // ═══════════════════════════════════════════════════════════════════════
    // PUT - Actualizar
    // ═══════════════════════════════════════════════════════════════════════

    /// <summary>
    /// Actualiza un/a {{Entity}} existente
    /// </summary>
    [HttpPut("{id:int}")]
    [Authorize(Roles = "Admin,Manager")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    [ProducesResponseType<ValidationProblemDetails>(StatusCodes.Status400BadRequest)]
    public async Task<IActionResult> Update(
        int id,
        [FromBody] Update{{Entity}}Request request,
        CancellationToken ct = default)
    {
        // El id solo viene de la ruta, asi que la discrepancia ruta/cuerpo ya no es representable.
        await _mediator.Send(_contratos.ToCommand(request) with { Id = id }, ct);

        _logger.LogInformation("{{Entity}} {Id} updated by {User}", id, User.Identity?.Name);

        return NoContent();
    }

    // ═══════════════════════════════════════════════════════════════════════
    // DELETE - Eliminar (borrado logico)
    // ═══════════════════════════════════════════════════════════════════════

    /// <summary>
    /// Elimina un/a {{Entity}} (soft delete)
    /// </summary>
    [HttpDelete("{id:int}")]
    [Authorize(Roles = "Admin")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public async Task<IActionResult> Delete(
        int id,
        CancellationToken ct = default)
    {
        await _mediator.Send(new Delete{{Entity}}Command(id), ct);

        _logger.LogInformation("{{Entity}} {Id} deleted by {User}", id, User.Identity?.Name);

        return NoContent();
    }
}

// ═══════════════════════════════════════════════════════════════════════════
