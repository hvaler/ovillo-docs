// ============================================================================
// TEMPLATE: CursorPage.cs - Pagina por keyset (una vez por proyecto)
// Skill: generador-crud (ADR-031, ADR-042)
// ============================================================================
// USO: Reemplazar {{Namespace}}. Se genera si no existe; si ya existe no se toca.
// Destino: src/Application/Common/Models/CursorPage.cs
// ============================================================================
//
// POR QUE EXISTE, ademas de PaginatedResult:
//
// PaginatedResult usa OFFSET (Skip/Take) y ademas hace COUNT(*). Las dos cosas se degradan
// LINEALMENTE con la profundidad: para devolver la pagina 5.000 de 20 filas el motor tiene que
// recorrer y descartar 100.000, y el COUNT recorre la tabla entera. En una tabla que crece, ese
// listado pasa de instantaneo a inusable sin que cambie una linea de codigo.
//
// Peor que lento: con OFFSET, si alguien inserta o borra una fila mientras el usuario pagina, las
// filas se DESPLAZAN. El usuario ve dos veces la misma o se salta una, y el bug no es reproducible.
//
// Keyset (o seek) no tiene ninguno de los dos problemas: pide "lo siguiente DESPUES de esta clave",
// que el motor resuelve con una busqueda en el indice, en tiempo constante respecto a la
// profundidad, y sobre un conjunto estable.
//
// CUANDO USAR CADA UNO:
//
//   PaginatedResult (offset)        CursorPage (keyset)
//   ---------------------------     ------------------------------------------------
//   El usuario salta a la pagina     Scroll infinito, "cargar mas", export por lotes,
//   N, o necesita ver "de 1.240"     sincronizacion, cualquier recorrido secuencial
//   Tablas pequenas y acotadas       Tablas que crecen sin techo
//   Rejillas de administracion       Listados de usuario final y APIs publicas
//
// Keyset NO da el total ni permite saltar a una pagina arbitraria: es el precio, y es la razon
// de que no sustituya al otro. Si la interfaz ensena "pagina 7 de 62", necesitas offset.

using System.Text;

namespace {{Namespace}}.Application.Common.Models;

/// <summary>
/// Pagina obtenida por keyset. <see cref="NextCursor"/> es null cuando no hay mas filas.
/// No lleva total: calcularlo exigiria el COUNT(*) que esta paginacion evita a proposito.
/// </summary>
public sealed record CursorPage<T>(IReadOnlyList<T> Items, string? NextCursor)
{
    public bool HasMore => NextCursor is not null;

    public static CursorPage<T> Empty { get; } = new(Array.Empty<T>(), null);
}

/// <summary>
/// Cursor opaco sobre la clave de orden (CreatedAt, Id). Se serializa en base64url para que viaje
/// en una query string sin escapar y para que el cliente no lo interprete ni lo construya a mano:
/// es un testigo, no una API.
/// </summary>
public readonly record struct Cursor(DateTimeOffset CreatedAt, int Id)
{
    public string Codificar()
    {
        // UtcTicks y no Ticks: el cursor tiene que dar el mismo valor venga de donde venga el offset.
        var crudo = $"{CreatedAt.UtcTicks}|{Id}";
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(crudo))
            .TrimEnd('=').Replace('+', '-').Replace('/', '_');
    }

    /// <summary>
    /// Decodifica un cursor recibido del cliente. Devuelve false ante cualquier cosa que no sea un
    /// cursor valido: llega de fuera, asi que se valida, nunca se confia.
    /// </summary>
    public static bool TryDecodificar(string? valor, out Cursor cursor)
    {
        cursor = default;
        if (string.IsNullOrWhiteSpace(valor)) return false;

        try
        {
            var b64 = valor.Replace('-', '+').Replace('_', '/');
            b64 = b64.PadRight(b64.Length + (4 - b64.Length % 4) % 4, '=');
            var partes = Encoding.UTF8.GetString(Convert.FromBase64String(b64)).Split('|');
            if (partes.Length != 2) return false;
            if (!long.TryParse(partes[0], out var ticks)) return false;
            if (!int.TryParse(partes[1], out var id)) return false;
            if (ticks < DateTimeOffset.MinValue.UtcTicks || ticks > DateTimeOffset.MaxValue.UtcTicks) return false;

            cursor = new Cursor(new DateTimeOffset(ticks, TimeSpan.Zero), id);
            return true;
        }
        catch (FormatException)
        {
            return false;
        }
    }
}
