// ============================================================================
// TEMPLATE: DbSession.cs - La unidad de trabajo con procedimientos (una vez por proyecto)
// Skill: generador-crud, acceso a datos por SP (ADR-053)
// ============================================================================
// USO: Reemplazar {{Namespace}}
// Destino: src/Infrastructure/Persistence/DbSession.cs
//
// QUE SUSTITUYE: al `ApplicationDbContext`, en su papel de IUnitOfWork. No en el de mapeador: aqui no
// hay mapeo porque el que sabe de columnas es el procedimiento.
//
// POR QUE IUnitOfWork SIGUE EXISTIENDO, que es la pregunta obvia: porque "unidad de trabajo" no
// significa "volcar el change tracker". Significa que un caso de uso que toca dos agregados se aplica
// entero o no se aplica -- la definicion de Fowler, anterior a EF Core y a Dapper. EF Core la
// implementa acumulando y volcando; aqui se implementa con una transaccion. El handler no cambia.
//
// COMO FUNCIONA: la conexion y la transaccion se abren PEREZOSAMENTE, en la primera operacion que las
// pida. Un caso de uso que solo lee nunca abre transaccion. Si el handler lanza antes de
// SaveChangesAsync, el Dispose deshace: no hay commit sin llamada explicita, que es justo la garantia
// que da el contrato.
//
// REGISTRO (Scoped, una por peticion):
//   services.AddScoped<DbSession>();
//   services.AddScoped<IDbSession>(sp => sp.GetRequiredService<DbSession>());
//   services.AddScoped<IUnitOfWork>(sp => sp.GetRequiredService<DbSession>());
// Las tres lineas apuntan a la MISMA instancia a proposito: si fueran tres, el repositorio escribiria
// en una transaccion y SaveChangesAsync confirmaria otra vacia.
// ============================================================================

using System.Data;
using Microsoft.Data.SqlClient;
using {{Namespace}}.Application.Common.Interfaces;
using {{Namespace}}.Domain.Entities;

namespace {{Namespace}}.Infrastructure.Persistence;

/// <summary>
/// Lo que los repositorios y las consultas piden para hablar con la base. Es de Infrastructure a
/// proposito: Application solo conoce <see cref="IUnitOfWork"/>, que es lo unico que necesita decir.
/// </summary>
public interface IDbSession
{
    /// <summary>Conexion abierta. La primera llamada la abre.</summary>
    Task<IDbConnection> ConexionAsync(CancellationToken ct = default);

    /// <summary>Transaccion en curso, o null si todavia nadie ha escrito.</summary>
    IDbTransaction? Transaccion { get; }

    /// <summary>Abre la transaccion si no habia. La llaman las operaciones de escritura, no las lecturas.</summary>
    Task EnEscrituraAsync(CancellationToken ct = default);

    /// <summary>Anota filas afectadas para que SaveChangesAsync devuelva el total, como el contrato promete.</summary>
    void Anotar(int filas);

    /// <summary>
    /// Rellena las marcas de auditoria. Es lo que en EF Core hace el interceptor de SaveChanges: aqui
    /// no hay interceptor porque no hay volcado, asi que lo hace quien escribe.
    /// </summary>
    void Auditar(IAuditable entidad, bool esAlta);
}

public sealed class DbSession : IDbSession, IUnitOfWork, IAsyncDisposable
{
    private readonly string _cadenaConexion;
    private SqlConnection? _conexion;
    private SqlTransaction? _transaccion;
    private int _filasAfectadas;

    public DbSession(string cadenaConexion)
    {
        _cadenaConexion = cadenaConexion;
    }

    public IDbTransaction? Transaccion => _transaccion;

    public async Task<IDbConnection> ConexionAsync(CancellationToken ct = default)
    {
        if (_conexion is null)
        {
            _conexion = new SqlConnection(_cadenaConexion);
            await _conexion.OpenAsync(ct);
        }

        return _conexion;
    }

    public async Task EnEscrituraAsync(CancellationToken ct = default)
    {
        var conexion = (SqlConnection)await ConexionAsync(ct);
        _transaccion ??= (SqlTransaction)await conexion.BeginTransactionAsync(ct);
    }

    public void Anotar(int filas) => _filasAfectadas += filas;

    public void Auditar(IAuditable entidad, bool esAlta)
    {
        var ahora = DateTimeOffset.UtcNow;

        if (esAlta)
        {
            entidad.CreatedAt = ahora;
        }
        else
        {
            entidad.ModifiedAt = ahora;
        }
    }

    /// <summary>
    /// Confirma la unidad de trabajo. Sin transaccion abierta no hay nada que confirmar y devuelve 0:
    /// es el caso de un caso de uso que solo ha leido, y no es un error.
    /// </summary>
    public async Task<int> SaveChangesAsync(CancellationToken cancellationToken = default)
    {
        if (_transaccion is null) return 0;

        await _transaccion.CommitAsync(cancellationToken);
        await _transaccion.DisposeAsync();
        _transaccion = null;

        var total = _filasAfectadas;
        _filasAfectadas = 0;
        return total;
    }

    /// <summary>
    /// Deshace lo no confirmado. Que el Dispose sea el que deshace es deliberado: significa que
    /// una excepcion a mitad de un caso de uso no deja media escritura, sin que el handler tenga que
    /// acordarse de nada.
    /// </summary>
    public async ValueTask DisposeAsync()
    {
        if (_transaccion is not null)
        {
            await _transaccion.RollbackAsync();
            await _transaccion.DisposeAsync();
        }

        if (_conexion is not null)
        {
            await _conexion.DisposeAsync();
        }
    }
}
