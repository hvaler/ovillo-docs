// ============================================================================
// TEMPLATE: Repository.cs - Repositorio del agregado con Dapper y SQL propio (por entidad)
// Skill: generador-crud, acceso a datos por SQL (ADR-055)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{Namespace}}, {{Schema}} y los fragmentos
//      {{ColumnasSp}}, {{ValoresSpCrear}}, {{ParametrosSpCrear_Valores}}, {{AsignacionesSpActualizar}},
//      {{ParametrosDapperCrear}}, {{ParametrosDapperActualizar}}, {{ImplementacionMetodosEspecificosSql}}
// Destino: src/Infrastructure/Persistence/Repositories/{{Entity}}Repository.cs
//
// La interfaz I{{Entity}}Repository es de DOMAIN y es LA MISMA que implementan los adaptadores de EF
// Core y de procedimientos. El caso de uso no cambia al cambiar de modo.
//
// EL MODELO DE ESCRITURA ES EL DE ADR-053: `AddAsync` y `UpdateAsync` ESCRIBEN al llamarlas, y mutar
// un agregado devuelto por `GetByIdAsync` no persiste nada. La transaccion la abre la sesion en la
// primera escritura y la confirma `IUnitOfWork.SaveChangesAsync`.
//
// LA SESION SE REUTILIZA del modo `dapper-sp` (ver compartidas.json): confirmar una unidad de trabajo
// no depende de donde viva el SQL.
//
// PARAMETRIZADO SIEMPRE: los valores viajan como parametros de Dapper (`@Name`), nunca interpolados.
// ============================================================================

using Dapper;
using {{Namespace}}.Domain.Entities;
using {{Namespace}}.Domain.Repositories;

namespace {{Namespace}}.Infrastructure.Persistence.Repositories;

public class {{Entity}}Repository : I{{Entity}}Repository
{
    /// <summary>Tope de GetAllAsync: sin el, una tabla que crece convierte el metodo en un incidente.</summary>
    private const int MaxSinPaginar = 200;

    // ─────────────────────────────────────────────────────────────────────── el SQL, en un solo sitio

    private const string SqlGet = """
        SELECT {{ColumnasSp}}
        FROM {{Schema}}.{{Entities}}
        WHERE Id = @Id;
        """;

    // El desempate por Id evita que una fila salga repetida en una pagina y falte en la siguiente.
    private const string SqlTodos = """
        SELECT TOP (@Tope) {{ColumnasSp}}
        FROM {{Schema}}.{{Entities}}
        ORDER BY CreatedAt DESC, Id ASC;
        """;

    private const string SqlCrear = """
        INSERT INTO {{Schema}}.{{Entities}} ({{ValoresSpCrear}}, CreatedAt)
        VALUES ({{ParametrosSpCrear_Valores}}, @CreatedAt);

        SELECT CAST(SCOPE_IDENTITY() AS INT);
        """;

    private const string SqlActualizar = """
        UPDATE {{Schema}}.{{Entities}}
        SET {{AsignacionesSpActualizar}},
            ModifiedAt = @ModifiedAt
        WHERE Id = @Id;

        SELECT @@ROWCOUNT;
        """;

    private const string SqlExiste = """
        SELECT CASE WHEN EXISTS (
            SELECT 1 FROM {{Schema}}.{{Entities}} WHERE Id = @Id
        ) THEN 1 ELSE 0 END;
        """;

    // ───────────────────────────────────────────────────────────────────────

    private readonly IDbSession _session;

    public {{Entity}}Repository(IDbSession session)
    {
        _session = session;
    }

    public async Task<{{Entity}}?> GetByIdAsync(int id, CancellationToken ct = default)
    {
        var conexion = await _session.ConexionAsync(ct);

        return await conexion.QuerySingleOrDefaultAsync<{{Entity}}>(new CommandDefinition(
            SqlGet, new { Id = id }, _session.Transaccion, cancellationToken: ct));
    }

    public async Task<IReadOnlyList<{{Entity}}>> GetAllAsync(CancellationToken ct = default)
    {
        var conexion = await _session.ConexionAsync(ct);

        return (await conexion.QueryAsync<{{Entity}}>(new CommandDefinition(
            SqlTodos, new { Tope = MaxSinPaginar }, _session.Transaccion, cancellationToken: ct))).ToList();
    }

    public async Task<{{Entity}}> AddAsync({{Entity}} entity, CancellationToken ct = default)
    {
        await _session.EnEscrituraAsync(ct);
        _session.Auditar(entity, esAlta: true);

        var conexion = await _session.ConexionAsync(ct);

        var id = await conexion.ExecuteScalarAsync<int>(new CommandDefinition(
            SqlCrear, {{ParametrosDapperCrear}}, _session.Transaccion, cancellationToken: ct));

        _session.Anotar(1);

        // El agregado recien creado no tiene Id -- lo asigna la base --, asi que se relee. Devolver la
        // instancia de entrada con Id 0 dejaria al handler registrando "created" con id 0 y al
        // controlador emitiendo una Location que no resuelve.
        return await GetByIdAsync(id, ct)
            ?? throw new InvalidOperationException(
                $"El INSERT devolvio el id {id} y la lectura posterior no encuentra esa fila.");
    }

    public async Task UpdateAsync({{Entity}} entity, CancellationToken ct = default)
    {
        await _session.EnEscrituraAsync(ct);
        _session.Auditar(entity, esAlta: false);

        var conexion = await _session.ConexionAsync(ct);
        var id = entity.Id;

        var filas = await conexion.ExecuteScalarAsync<int>(new CommandDefinition(
            SqlActualizar, {{ParametrosDapperActualizar}}, _session.Transaccion, cancellationToken: ct));

        _session.Anotar(filas);
    }

    public async Task<bool> ExistsAsync(int id, CancellationToken ct = default)
    {
        var conexion = await _session.ConexionAsync(ct);

        return await conexion.ExecuteScalarAsync<bool>(new CommandDefinition(
            SqlExiste, new { Id = id }, _session.Transaccion, cancellationToken: ct));
    }

    // ═══════════════════════════════════════════════════════════════════════
    // MÉTODOS ESPECÍFICOS
    // ═══════════════════════════════════════════════════════════════════════

    {{ImplementacionMetodosEspecificosSql}}
}
