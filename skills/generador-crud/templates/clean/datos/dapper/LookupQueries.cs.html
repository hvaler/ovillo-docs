// ============================================================================
// TEMPLATE: LookupQueries.cs - Catalogos cacheados con SQL propio (una vez por proyecto)
// Skill: generador-crud, acceso a datos por SQL (ADR-055)
// ============================================================================
// USO: Reemplazar {{Namespace}}, {{Schema}}
// Destino: src/Infrastructure/Persistence/Queries/CachedLookupQueries.cs
// Registro: services.AddMemoryCache();
//           services.AddScoped<ILookupQueries<ScholarshipType>, CachedLookupQueries<ScholarshipType>>();
//
// LA TABLA SE DERIVA DEL TIPO: `{{Schema}}.{T}s`. Es una convencion, y es deliberada: la interfaz es
// generica y el generico no puede saber el nombre de otra forma. Es lo mismo que hace EF Core con
// `Set<T>()`, solo que dicho en voz alta. Si un catalogo no la sigue, hereda esta clase y sobrescribe
// `Tabla`.
//
// LO QUE SE CONCATENA ES `typeof(T).Name`, un nombre de tipo del propio ensamblado, nunca algo que
// venga de fuera: no hay superficie de inyeccion. El resto -- el codigo y el id -- son parametros.
//
// LAS COLUMNAS SON FIJAS porque `LookupEntity` las fija: todo catalogo tiene las mismas. Por eso esta
// plantilla no necesita un fragmento por entidad, a diferencia de las de agregado.
// ============================================================================

using Dapper;
using Microsoft.Extensions.Caching.Memory;
using {{Namespace}}.Application.Common.Interfaces;
using {{Namespace}}.Domain.Entities;

namespace {{Namespace}}.Infrastructure.Persistence.Queries;

public class CachedLookupQueries<T> : ILookupQueries<T> where T : LookupEntity
{
    private static readonly TimeSpan Duracion = TimeSpan.FromMinutes(30);

    private readonly IDbSession _session;
    private readonly IMemoryCache _cache;

    public CachedLookupQueries(IDbSession session, IMemoryCache cache)
    {
        _session = session;
        _cache = cache;
    }

    /// <summary>Tabla del catalogo. Sobrescribible por si no sigue la convencion.</summary>
    protected virtual string Tabla => $"{{Schema}}.{typeof(T).Name}s";

    public async Task<IReadOnlyList<T>> GetAllActiveAsync(CancellationToken ct = default)
    {
        var clave = $"lookup:{typeof(T).Name}";

        if (_cache.TryGetValue(clave, out IReadOnlyList<T>? cacheado) && cacheado is not null)
        {
            return cacheado;
        }

        var conexion = await _session.ConexionAsync(ct);

        var sql = $"""
            SELECT Id, Code, Name, Description, IsActive, SortOrder
            FROM {Tabla}
            WHERE IsActive = 1
            ORDER BY SortOrder, Name;
            """;

        var filas = (await conexion.QueryAsync<T>(new CommandDefinition(
            sql, transaction: _session.Transaccion, cancellationToken: ct))).ToList();

        _cache.Set(clave, (IReadOnlyList<T>)filas, Duracion);
        return filas;
    }

    // Los dos accesos por clave se resuelven SOBRE LA LISTA CACHEADA, no con otra consulta. Un
    // catalogo cabe en memoria por definicion -- si no cupiera no seria un catalogo --, asi que ir a
    // la base para buscar un elemento de una lista que ya se tiene delante seria una ida de mas.
    public async Task<T?> GetByCodeAsync(string code, CancellationToken ct = default)
    {
        var todos = await GetAllActiveAsync(ct);
        return todos.FirstOrDefault(e => e.Code == code);
    }

    public async Task<T?> GetByIdAsync(int id, CancellationToken ct = default)
    {
        var todos = await GetAllActiveAsync(ct);
        return todos.FirstOrDefault(e => e.Id == id);
    }
}
