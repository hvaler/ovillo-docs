// TEMPLATE: Create{{Entity}}.cs - un caso de uso completo con procedimientos (por entidad)
// Skill: generador-crud, estilo vertical-slices con acceso por SP (ADR-054)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{entities}}, {{Namespace}}, {{Schema}} y los fragmentos
//      {{PropiedadesCreateRequest}}, {{ReglasValidacionSincronas}}, {{ComprobacionUnicidadSp}},
//      {{ArgumentosCrear}}, {{ParametrosDapperCrear}}
// Destino: src/Features/{{Entities}}/Create{{Entity}}.cs
//
// Peticion, validacion, handler y endpoint EN UN FICHERO: eso es la slice. Lo unico que cambia
// respecto al modo EF Core son las diez lineas del handler que hablan con la base.
// ============================================================================

using System.Data;
using Dapper;
using FluentValidation;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Routing;
using {{Namespace}}.Common;
using {{Namespace}}.Domain.Entities;

namespace {{Namespace}}.Features.{{Entities}};

public sealed record Create{{Entity}}Request
{
    {{PropiedadesCreateRequest}}
}

public sealed record Create{{Entity}}Response(int Id);

public sealed class Create{{Entity}}Validator : AbstractValidator<Create{{Entity}}Request>
{
    public Create{{Entity}}Validator()
    {
        {{ReglasValidacionSincronas}}
    }
}

public sealed class Create{{Entity}}Handler
{
    private readonly IDbConnection _db;

    public Create{{Entity}}Handler(IDbConnection db) => _db = db;

    public async Task<Result<Create{{Entity}}Response>> Handle(
        Create{{Entity}}Request request,
        CancellationToken ct = default)
    {
        var validacion = await new Create{{Entity}}Validator().ValidateAsync(request, ct);
        if (!validacion.IsValid)
            return Result<Create{{Entity}}Response>.Invalid(
                validacion.Errors.Select(e => e.ErrorMessage).ToList());

        // La unicidad se comprueba AQUI y no en el validador: el handler ya tiene la conexion delante
        // y un Result que devolver. Un choque es Conflict, no Validation.
        {{ComprobacionUnicidadSp}}

        // La entidad se construye igual que con EF Core: la factoria del dominio valida las reglas de
        // negocio, y eso no depende de como se guarde despues.
        var entity = {{Entity}}.Create({{ArgumentosCrear}});

        var id = await _db.ExecuteScalarAsync<int>(new CommandDefinition(
            "{{Schema}}.Create{{Entity}}",
            {{ParametrosDapperCrear}},
            commandType: CommandType.StoredProcedure,
            cancellationToken: ct));

        return Result<Create{{Entity}}Response>.Ok(new Create{{Entity}}Response(id));
    }
}

public static class Create{{Entity}}Endpoint
{
    public static IEndpointRouteBuilder MapCreate{{Entity}}(this IEndpointRouteBuilder rutas)
    {
        rutas.MapPost("/api/v1/{{entities}}", async (
            Create{{Entity}}Request request,
            Create{{Entity}}Handler handler,
            CancellationToken ct) =>
        {
            var resultado = await handler.Handle(request, ct);
            return resultado.Succeeded
                ? Results.Created($"/api/v1/{{entities}}/{resultado.Value!.Id}", resultado.Value)
                : resultado.ToProblem();
        })
        .WithName("Create{{Entity}}")
        .Produces<Create{{Entity}}Response>(StatusCodes.Status201Created)
        .ProducesValidationProblem();

        return rutas;
    }
}
