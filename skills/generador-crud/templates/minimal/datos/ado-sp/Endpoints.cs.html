// TEMPLATE: {{Entity}}Endpoints.cs - el CRUD entero con ADO.NET puro y procedimientos (por entidad)
// Skill: generador-crud, estilo minimal con acceso por ADO.NET (ADR-056)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{entities}}, {{Namespace}}, {{Schema}} y los fragmentos
//      {{PropiedadesCreateRequest}}, {{LecturaReaderMinimal}}, {{ParametrosAdoCrearRequest}},
//      {{ParametrosAdoActualizarRequest}}
// Destino: src/Endpoints/{{Entity}}Endpoints.cs
//
// NI ORM NI MICRO-ORM, y el precio esta a la vista: este fichero es el doble de largo que el mismo
// CRUD con Dapper, y lo que ocupa la diferencia es codigo que Dapper escribe.
//
// LO QUE ESTE MODO EVITA ES UN PAQUETE DE TERCEROS, no las dependencias. Sigue necesitando
// `Microsoft.Data.SqlClient`, que es mayor que Dapper: lo unico que se ahorra es Dapper. La
// restriccion a la que sirve es «nada fuera de Microsoft», que en algunas casas es una politica
// real, o un codigo heredado que ya es ADO.NET y al que el codigo nuevo tiene que parecerse. Son
// dos de los cuatro casos que la guia de acceso a datos de la organizacion recoge.
//
// NO ES POR RENDIMIENTO: esa misma guia mide la diferencia con Dapper en menos del 5 %. Si la
// razon para elegir este modo fuera la velocidad, la razon seria mala.
//
// LOS PARAMETROS VAN TIPADOS, nunca con `AddWithValue`: ese infiere el tipo del valor de C#, manda una
// `string` como `nvarchar(4000)` y, si la columna es `varchar(200)`, SQL Server convierte, el indice
// deja de servir y la consulta pasa de buscar a escanear. No da sintoma hasta que la tabla crece.
//
// LAS COLUMNAS SE LEEN POR NOMBRE (`GetOrdinal`), no por posicion: anadir una columna al procedimiento
// desplaza las demas y el codigo seguiria compilando, leyendo la fecha donde estaba el nombre.
// ============================================================================

using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Routing;
using Microsoft.Data.SqlClient;
using {{Namespace}}.Data;

namespace {{Namespace}}.Endpoints;

    {{PropiedadesCreateRequest}}
}

public static class {{Entity}}Endpoints
{
    /// <summary>Materializa una fila. Por nombre de columna, nunca por posicion.</summary>
    private static {{Entity}} Leer(SqlDataReader reader)
    {
        {{LecturaReaderMinimal}}
    }

    /// <summary>Un procedimiento con su conexion abierta. Todo lo de abajo empieza por aqui.</summary>
    private static async Task<(SqlConnection Cn, SqlCommand Cmd)> AbrirAsync(
        SqlConnection cn, string procedimiento, CancellationToken ct)
    {
        await cn.OpenAsync(ct);

        var cmd = cn.CreateCommand();
        cmd.CommandText = procedimiento;
        cmd.CommandType = CommandType.StoredProcedure;

        return (cn, cmd);
    }

    public static IEndpointRouteBuilder Map{{Entities}}(this IEndpointRouteBuilder rutas)
    {
        var grupo = rutas.MapGroup("/api/v1/{{entities}}");

        grupo.MapGet("/", async (SqlConnection conexion, int page = 1, int size = 20, CancellationToken ct = default) =>
        {
            var (cn, cmd) = await AbrirAsync(conexion, "{{Schema}}.List{{Entities}}", ct);
            await using (cmd)
            {
                // El tope lo aplica el procedimiento, que es superficie publica: cualquiera puede
                // invocarlo sin pasar por esta ruta.
                cmd.Parameters.Add("@Page", SqlDbType.Int).Value = page;
                cmd.Parameters.Add("@Size", SqlDbType.Int).Value = size;

                await using var reader = await cmd.ExecuteReaderAsync(ct);

                // Dos conjuntos -- la pagina y el total -- recorridos con NextResultAsync: una ida a
                // la base, y el total corresponde a la misma foto que la pagina.
                var items = new List<{{Entity}}>();
                while (await reader.ReadAsync(ct)) items.Add(Leer(reader));

                var total = 0;
                if (await reader.NextResultAsync(ct) && await reader.ReadAsync(ct))
                {
                    total = reader.GetInt32(0);
                }

                return Results.Ok(new { items, total, page, size });
            }
        });

        grupo.MapGet("/{id:int}", async (int id, SqlConnection conexion, CancellationToken ct) =>
        {
            var (cn, cmd) = await AbrirAsync(conexion, "{{Schema}}.Get{{Entity}}", ct);
            await using (cmd)
            {
                cmd.Parameters.Add("@Id", SqlDbType.Int).Value = id;

                await using var reader = await cmd.ExecuteReaderAsync(ct);
                return await reader.ReadAsync(ct) ? Results.Ok(Leer(reader)) : Results.NotFound();
            }
        });

        grupo.MapPost("/", async (Create{{Entity}}Request request, SqlConnection conexion, CancellationToken ct) =>
        {
            var (cn, cmd) = await AbrirAsync(conexion, "{{Schema}}.Create{{Entity}}", ct);
            await using (cmd)
            {
                {{ParametrosAdoCrearRequest}}

                var id = Convert.ToInt32(await cmd.ExecuteScalarAsync(ct));
                return Results.Created($"/api/v1/{{entities}}/{id}", new { id });
            }
        });

        grupo.MapPut("/{id:int}", async (int id, Create{{Entity}}Request request, SqlConnection conexion, CancellationToken ct) =>
        {
            var (cn, cmd) = await AbrirAsync(conexion, "{{Schema}}.Update{{Entity}}", ct);
            await using (cmd)
            {
                cmd.Parameters.Add("@Id", SqlDbType.Int).Value = id;
                {{ParametrosAdoActualizarRequest}}

                // El @@ROWCOUNT dice si existia: con esto no hace falta leer antes para el 404.
                var filas = Convert.ToInt32(await cmd.ExecuteScalarAsync(ct));
                return filas > 0 ? Results.NoContent() : Results.NotFound();
            }
        });

        grupo.MapDelete("/{id:int}", async (int id, SqlConnection conexion, CancellationToken ct) =>
        {
            var (cn, cmd) = await AbrirAsync(conexion, "{{Schema}}.Delete{{Entity}}", ct);
            await using (cmd)
            {
                cmd.Parameters.Add("@Id", SqlDbType.Int).Value = id;

                var filas = Convert.ToInt32(await cmd.ExecuteScalarAsync(ct));
                return filas > 0 ? Results.NoContent() : Results.NotFound();
            }
        });

        return rutas;
    }
}
