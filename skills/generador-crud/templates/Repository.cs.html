// TEMPLATE: Repository.cs - Repositorio con EF Core (por entidad)
// Skill: generador-crud (ADR-031)
// ============================================================================
// USO: Reemplazar {{Entity}}, {{Entities}}, {{Namespace}} y el fragmento {{ImplementacionMetodosEspecificos}}
// Destino: src/Infrastructure/Persistence/Repositories/{{Entity}}Repository.cs
// La interfaz I{{Entity}}Repository es de Application (plantilla IRepository.cs).
// ============================================================================

using Microsoft.EntityFrameworkCore;
using {{Namespace}}.Application.Common.Interfaces;
using {{Namespace}}.Application.Common.Models;
using {{Namespace}}.Domain.Entities;

namespace {{Namespace}}.Infrastructure.Persistence.Repositories;

public class {{Entity}}Repository : I{{Entity}}Repository
{
    /// <summary>Tope de GetAllAsync: sin el, una tabla que crece convierte el metodo en un incidente.</summary>
    private const int MaxSinPaginar = 1000;
    private const int MaxPorPagina = 200;

    private readonly ApplicationDbContext _context;

    public {{Entity}}Repository(ApplicationDbContext context)
    {
        _context = context;
    }

    public async Task<{{Entity}}?> GetByIdAsync(int id, CancellationToken ct = default)
    {
        return await _context.{{Entities}}
            .FirstOrDefaultAsync(e => e.Id == id, ct);
    }

    public async Task<PaginatedResult<{{Entity}}>> GetPaginatedAsync(
        int page,
        int size,
        CancellationToken ct = default)
    {
        // page = 0 producia Skip(-size), que lanza en ejecucion; y sin desempate, dos filas con el mismo
        // CreatedAt pueden salir repetidas en una pagina y faltar en la siguiente (ADR-027).
        page = Math.Max(1, page);
        size = Math.Clamp(size, 1, MaxPorPagina);

        var query = _context.{{Entities}}.AsNoTracking();

        var totalCount = await query.CountAsync(ct);

        var items = await query
            .OrderByDescending(e => e.CreatedAt)
            .ThenBy(e => e.Id)
            .Skip((page - 1) * size)
            .Take(size)
            .ToListAsync(ct);

        return new PaginatedResult<{{Entity}}>(items, totalCount, page, size);
    }

    public async Task<IReadOnlyList<{{Entity}}>> GetAllAsync(CancellationToken ct = default)
    {
        return await _context.{{Entities}}
            .AsNoTracking()
            .OrderByDescending(e => e.CreatedAt)
            .ThenBy(e => e.Id)
            .Take(MaxSinPaginar)
            .ToListAsync(ct);
    }

    // AddAsync solo existe para generadores de valor que consultan la base (HiLo). Con Id identity es Add,
    // y el trabajo asincrono es SaveChangesAsync, que llama el handler a traves de IUnitOfWork.
    public {{Entity}} Add({{Entity}} entity)
    {
        _context.{{Entities}}.Add(entity);
        return entity;
    }

    // Update marca el grafo completo como modificado y fuerza un UPDATE de todas las columnas. Sobre una
    // entidad ya rastreada (la que devuelve GetByIdAsync) no hace falta: el change tracker ya sabe que cambio.
    public void Update({{Entity}} entity)
    {
        if (_context.Entry(entity).State == EntityState.Detached)
        {
            _context.{{Entities}}.Update(entity);
        }
    }

    public async Task<bool> ExistsAsync(int id, CancellationToken ct = default)
    {
        return await _context.{{Entities}}
            .AnyAsync(e => e.Id == id, ct);
    }

    // ═══════════════════════════════════════════════════════════════════════

    // ═══════════════════════════════════════════════════════════════════════

    {{ImplementacionMetodosEspecificos}}
}
