using Mediator;
using Microsoft.Extensions.Logging;
using {{Namespace}}.Application.Common.Exceptions;
using {{Namespace}}.Application.Common.Interfaces;
using {{Namespace}}.Application.Features.{{Entities}}.Dtos;
using {{Namespace}}.Domain.Entities;

namespace {{Namespace}}.Application.Features.{{Entities}}.Commands;

public sealed class Create{{Entity}}CommandHandler : ICommandHandler<Create{{Entity}}Command, {{Entity}}Dto>
{
    private readonly I{{Entity}}Repository _repository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly {{Entity}}Mapper _mapper;
    private readonly ILogger<Create{{Entity}}CommandHandler> _logger;

    public Create{{Entity}}CommandHandler(
        I{{Entity}}Repository repository,
        IUnitOfWork unitOfWork,
        {{Entity}}Mapper mapper,
        ILogger<Create{{Entity}}CommandHandler> logger)
    {
        _repository = repository;
        _unitOfWork = unitOfWork;
        _mapper = mapper;
        _logger = logger;
    }

    public async ValueTask<{{Entity}}Dto> Handle(Create{{Entity}}Command request, CancellationToken cancellationToken)
    {
        // La factoria del dominio valida las reglas de negocio y lanza DomainException si no se cumplen
        var entity = {{Entity}}.Create(
            {{ArgumentosCrear}}
        );

        _repository.Add(entity);
        await _unitOfWork.SaveChangesAsync(cancellationToken);

        _logger.LogInformation("{{Entity}} {Id} created", entity.Id);

        return _mapper.ToDto(entity);
    }
}

public sealed class Update{{Entity}}CommandHandler : ICommandHandler<Update{{Entity}}Command>
{
    private readonly I{{Entity}}Repository _repository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly ILogger<Update{{Entity}}CommandHandler> _logger;

    public Update{{Entity}}CommandHandler(
        I{{Entity}}Repository repository,
        IUnitOfWork unitOfWork,
        ILogger<Update{{Entity}}CommandHandler> logger)
    {
        _repository = repository;
        _unitOfWork = unitOfWork;
        _logger = logger;
    }

    public async ValueTask<Unit> Handle(Update{{Entity}}Command request, CancellationToken cancellationToken)
    {
        var entity = await _repository.GetByIdAsync(request.Id, cancellationToken)
            ?? throw new NotFoundException(nameof({{Entity}}), request.Id);

        // La entidad viene rastreada: basta modificarla y guardar
        entity.Update(
            {{ArgumentosActualizar}}
        );

        await _unitOfWork.SaveChangesAsync(cancellationToken);

        _logger.LogInformation("{{Entity}} {Id} updated", entity.Id);
        return Unit.Value;
    }
}

public sealed class Delete{{Entity}}CommandHandler : ICommandHandler<Delete{{Entity}}Command>
{
    private readonly I{{Entity}}Repository _repository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly ICurrentUserService _currentUser;
    private readonly ILogger<Delete{{Entity}}CommandHandler> _logger;

    public Delete{{Entity}}CommandHandler(
        I{{Entity}}Repository repository,
        IUnitOfWork unitOfWork,
        ICurrentUserService currentUser,
        ILogger<Delete{{Entity}}CommandHandler> logger)
    {
        _repository = repository;
        _unitOfWork = unitOfWork;
        _currentUser = currentUser;
        _logger = logger;
    }

    public async ValueTask<Unit> Handle(Delete{{Entity}}Command request, CancellationToken cancellationToken)
    {
        var entity = await _repository.GetByIdAsync(request.Id, cancellationToken)
            ?? throw new NotFoundException(nameof({{Entity}}), request.Id);

        // Borrado logico con el metodo de dominio (la base SoftDeleteEntity lo hace idempotente)
        entity.SoftDelete(_currentUser.UserName ?? "system");

        await _unitOfWork.SaveChangesAsync(cancellationToken);

        _logger.LogInformation("{{Entity}} {Id} deleted by {User}", entity.Id, _currentUser.UserName);
        return Unit.Value;
    }
}
