# Actualizaciones automaticas de dependencias. Skill: git-best-practices
# Ubicacion: .github/dependabot.yml
# Complementa a nugets-management (auditoria, consolidacion, CPM): Dependabot abre los PRs con
# Conventional Commits y el ruleset de main exige que pasen los checks antes de mergear.
version: 2
updates:
  # Paquetes NuGet de la solucion (src/ en el layout Ovillo). Con Central Package Management basta
  # Directory.Packages.props: Dependabot lo entiende.
  - package-ecosystem: "nuget"
    directory: "/src"
    schedule:
      interval: "weekly"
      day: "monday"
      time: "07:00"
      timezone: "Europe/Madrid"
    open-pull-requests-limit: 5
    labels: ["dependencies", "nuget"]
    commit-message:
      prefix: "chore(deps)"
    groups:
      # Un PR para los minors y parches de Microsoft (EF Core, ASP.NET Core, extensiones)
      microsoft:
        patterns: ["Microsoft.*", "System.*"]
        update-types: ["minor", "patch"]
      # Un PR para el resto de minors y parches; los majors van uno a uno
      minor-y-patch:
        patterns: ["*"]
        exclude-patterns: ["Microsoft.*", "System.*"]
        update-types: ["minor", "patch"]
    # Para fijar un mayor que se decidio no adoptar (y registrarlo en _hilo/DECISIONES.md):
    # ignore:
    #   - dependency-name: "FluentAssertions"
    #     versions: [">=8.0.0"]

  # Las propias acciones de los workflows. Van pinneadas por SHA (templates/github-workflow.yml.template):
  # Dependabot actualiza el SHA y el comentario de version a la vez.
  - package-ecosystem: "github-actions"
    directory: "/"
    schedule:
      interval: "weekly"
      day: "monday"
      time: "07:00"
      timezone: "Europe/Madrid"
    labels: ["dependencies", "ci"]
    commit-message:
      prefix: "ci(deps)"
    groups:
      actions:
        patterns: ["*"]
