#!/usr/bin/env node
// init.js - Scaffold del proyecto consumidor para el plugin hv.
//
// Escribe en el repo del proyecto TODO lo que el plugin NO puede llevar dentro: el layout (src/docs/ops),
// el Hilo vacio de arranque, el Patron entero, el CLAUDE.md del consumidor con sus @imports, los ficheros
// del ecosistema que no son skills/agents/hooks (CLAUDE_BASE, esquemas, statusline, smoke-tests), las 3
// reglas incondicionales y el settings.json del consumidor (statusLine + anuncios: un plugin no puede
// llevarlos). Idempotente: NUNCA sobrescribe lo que ya existe, salvo con --force sobre ficheros concretos.
//
// Uso (lo invoca la skill /hv:init):
//   node "${CLAUDE_PLUGIN_ROOT}/skills/init/scripts/init.js" [--dest <dir>] [--ide none|visual-studio|vscode|rider] [--refresh-memoria] [--dry-run]
//
// Origen del andamio: ${CLAUDE_PLUGIN_ROOT}/scaffold/ (lo genera build-plugin.js desde ovillo/).

'use strict';
const fs = require('fs');
const path = require('path');

const args = process.argv.slice(2);

// Un argumento que no se reconoce NO puede acabar en una escritura real (ADR-036): antes se ignoraba, asi que
// `init.js --help` escribia el andamio entero en el repositorio del usuario. --help sale 0; lo desconocido, 2.
const USO = [
  'uso: init.js [--dest <dir>] [--ide none|visual-studio|vscode|rider] [--refresh-memoria] [--dry-run]',
  '',
  '  --dest <dir>        raiz del proyecto donde escribir (por defecto CLAUDE_PROJECT_DIR o el directorio actual)',
  '  --ide <valor>       integracion de IDE: none (por defecto), visual-studio, vscode, rider',
  '  --refresh-memoria   ademas, reescribe lo que es del ecosistema y no se personaliza (reglas, CLAUDE_BASE, CONFIGURACION)',
  '  --dry-run           no escribe nada: lista lo que haria',
  '  --help, -h          esta ayuda',
  '',
  'Idempotente: nunca sobrescribe lo que ya existe (salvo --refresh-memoria sobre ficheros del ecosistema).'
].join('\n');
{
  const CON_VALOR = new Set(['--dest', '--ide']);
  const SOLOS = new Set(['--dry-run', '--refresh-memoria', '--help', '-h']);
  const desconocidos = [];
  for (let i = 0; i < args.length; i++) {
    const a = args[i];
    if (CON_VALOR.has(a)) { i++; continue; }        // su valor no es un argumento suelto
    if (SOLOS.has(a)) continue;
    desconocidos.push(a);
  }
  if (args.includes('--help') || args.includes('-h')) { console.log(USO); process.exit(0); }
  if (desconocidos.length) {
    console.error('[hv] init: argumento no reconocido: ' + desconocidos.join(' ') + '. No se ha escrito nada.');
    console.error(USO);
    process.exit(2);
  }
  for (const k of CON_VALOR) {
    const i = args.indexOf(k);
    if (i >= 0 && (!args[i + 1] || args[i + 1].startsWith('-'))) {
      console.error('[hv] init: ' + k + ' necesita un valor. No se ha escrito nada.');
      console.error(USO);
      process.exit(2);
    }
  }
  const ideVal = (() => { const i = args.indexOf('--ide'); return i >= 0 ? args[i + 1] : null; })();
  if (ideVal && !['none', 'visual-studio', 'vscode', 'rider'].includes(ideVal)) {
    console.error('[hv] init: --ide "' + ideVal + '" no es un valor valido (none|visual-studio|vscode|rider). No se ha escrito nada.');
    process.exit(2);
  }
}

const opt = (k, d) => { const i = args.indexOf(k); return i >= 0 && args[i + 1] ? args[i + 1] : d; };
const DRY = args.includes('--dry-run');
const REFRESH = args.includes('--refresh-memoria');  // opt-in: refresca la memoria del ecosistema (ADR-003)
const DEST = path.resolve(opt('--dest', process.env.CLAUDE_PROJECT_DIR || process.cwd()));
const IDE = opt('--ide', null);
const PLUGIN_ROOT = process.env.CLAUDE_PLUGIN_ROOT || path.resolve(__dirname, '..', '..', '..');
const SCAFFOLD = path.join(PLUGIN_ROOT, 'scaffold');

if (!fs.existsSync(SCAFFOLD)) {
  console.error('[hv] init: no encuentro el andamio en ' + SCAFFOLD + '. Reinstala el plugin (claude plugin install hv@ovillo).');
  process.exit(1);
}

const hecho = { dirs: 0, files: 0, kept: 0 };
const log = [];
// Destinos ya previstos en esta pasada: en --dry-run nada se escribe, asi que un fichero o una carpeta que se
// tocan dos veces en la misma pasada (un README que entra por el bucle y por el copyTree de su carpeta; una
// subcarpeta que declara el manifiesto y que copyTree vuelve a recorrer) se contaban dos veces, y el simulacro
// anunciaba mas de lo que la pasada real escribe (ADR-036 D4). Las carpetas se guardan con '/' al final.
const previstos = new Set();
function mkdir(rel) {
  const p = path.join(DEST, rel);
  if (fs.existsSync(p) || (DRY && previstos.has(rel + '/'))) return;
  if (!DRY) fs.mkdirSync(p, { recursive: true });
  previstos.add(rel + '/');
  hecho.dirs++; log.push('  + ' + rel + '/');
}
function copy(relSrc, relDst, { force = false } = {}) {
  const src = path.join(SCAFFOLD, relSrc);
  const rel = relDst || relSrc;
  const dst = path.join(DEST, rel);
  if (!fs.existsSync(src)) { log.push('  ! falta en el plugin: ' + relSrc); return; }
  if ((fs.existsSync(dst) || (DRY && previstos.has(rel))) && !force) { hecho.kept++; return; }
  if (!DRY) { fs.mkdirSync(path.dirname(dst), { recursive: true }); fs.copyFileSync(src, dst); }
  previstos.add(rel);
  hecho.files++; log.push('  + ' + rel);
}
function copyTree(relSrc, relDst) {
  const base = path.join(SCAFFOLD, relSrc);
  if (!fs.existsSync(base)) { log.push('  ! falta en el plugin: ' + relSrc + '/'); return; }
  (function walk(d) {
    for (const e of fs.readdirSync(d, { withFileTypes: true })) {
      const p = path.join(d, e.name);
      const rel = path.relative(base, p).split(path.sep).join('/');
      if (e.isDirectory()) { mkdir(path.posix.join(relDst || relSrc, rel)); walk(p); }
      else copy(path.posix.join(relSrc, rel), path.posix.join(relDst || relSrc, rel));
    }
  })(base);
}
function writeIfMissing(rel, content) {
  const p = path.join(DEST, rel);
  if (fs.existsSync(p) || (DRY && previstos.has(rel))) { hecho.kept++; return; }
  previstos.add(rel);
  if (!DRY) { fs.mkdirSync(path.dirname(p), { recursive: true }); fs.writeFileSync(p, content, 'utf8'); }
  hecho.files++; log.push('  + ' + rel);
}

// 1) layout del consumidor, tal como lo declara el manifiesto del andamio
let manifiesto = null;
try { manifiesto = JSON.parse(fs.readFileSync(path.join(SCAFFOLD, '.claude', 'estructura-plantilla.json'), 'utf8').replace(/^﻿/, '')); } catch { /* sin manifiesto: layout minimo */ }
const dirs = manifiesto
  ? [].concat(manifiesto.directorios.siempre || [], manifiesto.directorios.solo_instalacion || [])
  : ['_hilo', '_hilo/specs', '_hilo/reuniones', '_patron', 'src', 'docs', 'ops'];
for (const d of dirs) mkdir(d);

// 2) README de cada carpeta del layout + ficheros de arranque de la raiz (nunca pisan)
for (const d of ['src', 'docs', 'docs/management', 'docs/architecture', 'docs/testing', 'docs/support', 'ops', 'ops/environment', 'ops/cicd', 'ops/testing']) copy(d + '/README.md');
copyTree('docs/management');            // config_proyecto.json, checklist, plantillas de requisitos y actas
copyTree('ops/environment');            // docker-compose.yml, .env.example
for (const f of ['CLAUDE.md', 'README.md', 'INICIO_RAPIDO.md', 'ecosystem.config.example.json', 'VERSION.json', '.gitattributes', '.gitignore']) copy(f);

// 3) Hilo (memoria vacia de arranque) y Patron (base de conocimiento entera)
copyTree('_hilo');
copyTree('_patron');

// 4) lo del ecosistema que un plugin no puede llevar como skill/agent/hook
mkdir('.claude'); mkdir('.claude/rules');
for (const f of ['CLAUDE_BASE.md', 'CLAUDE_BASE_EXTENDIDO.md', 'CONFIGURACION.md', 'statusline.js',
                 'estructura-plantilla.json', 'estructura-solucion.json', 'user-config.template.json']) copy('.claude/' + f);
copyTree('.claude/schemas'); copyTree('.claude/scripts'); copyTree('.claude/smoke-tests');
copyTree('.claude/rules');              // las 3 incondicionales: el runtime no carga rules/ desde un plugin
copy('.claude/settings.json');          // statusLine + anuncios; SIN hooks (los trae el plugin)

// 4-bis) --refresh-memoria (opt-in, ADR-003): pone al dia lo que es del ecosistema y NO se personaliza,
// y en el CLAUDE.md del proyecto solo retira el @import de CLAUDE_BASE (que dejo de estar siempre cargado).
// Lo personalizado (glosario, informacion del proyecto, reglas propias) no se toca nunca: se deja al lado
// una copia de la plantilla vigente para que el usuario compare y pode a mano lo que ya vive en skills.
if (REFRESH) {
  const propias = ['.claude/CLAUDE_BASE.md', '.claude/CLAUDE_BASE_EXTENDIDO.md', '.claude/CONFIGURACION.md'];
  for (const rel of propias) copy(rel, rel, { force: true });
  const dirReglas = path.join(SCAFFOLD, '.claude', 'rules');
  if (fs.existsSync(dirReglas)) {
    for (const f of fs.readdirSync(dirReglas).sort()) copy('.claude/rules/' + f, '.claude/rules/' + f, { force: true });
  }
  const memoria = path.join(DEST, 'CLAUDE.md');
  const plantilla = path.join(SCAFFOLD, 'CLAUDE.md');
  if (fs.existsSync(memoria) && fs.existsSync(plantilla)) {
    const antes = fs.readFileSync(memoria, 'utf8');
    const nota = '> Los estandares de la organizacion viven en `.claude/CLAUDE_BASE.md` y ya no se importan siempre:\n'
               + '> la regla `estandares-base` los recuerda al tocar codigo de `src/`.\n';
    const despues = antes.replace(/^@\.claude\/CLAUDE_BASE\.md[ \t]*\r?\n/m, nota);
    if (despues !== antes) {
      if (!DRY) fs.writeFileSync(memoria, despues, 'utf8');
      let ahorro = 0;
      try { ahorro = fs.statSync(path.join(DEST, '.claude', 'CLAUDE_BASE.md')).size; } catch { /* sin fichero: sin cifra */ }
      log.push('  ~ CLAUDE.md: retirado el @import de CLAUDE_BASE' + (ahorro ? ' (~' + Math.round(ahorro / 4) + ' tok menos siempre cargados)' : ''));
    } else {
      log.push('  = CLAUDE.md: ya no importaba CLAUDE_BASE, sin cambios');
    }
    const copia = path.join(DEST, 'CLAUDE.md.plantilla');
    if (!DRY) fs.writeFileSync(copia, fs.readFileSync(plantilla, 'utf8'), 'utf8');
    log.push('  + CLAUDE.md.plantilla: plantilla vigente para comparar y podar a mano (borrala al acabar)');
  }
}

// 4-ter) .gitignore del proyecto: NUNCA se pisa (paso 2), asi que en un repo que ya lo tenia faltaban las entradas
// locales del ecosistema y aparecian como untracked (p. ej. _hilo/.skill-guard.json, que escribe el hook skill-guard).
// Se anaden las que falten al final, con su cabecera, sin tocar el resto (ADR-035).
{
  const dst = path.join(DEST, '.gitignore');
  const necesarias = ['_hilo/.mcp-credentials.json', '_hilo/.update-check.json', '_hilo/.skill-guard.json',
                      '.claude/settings.local.json', '.claude/user-config.json', '.claude/sql-legacy-baseline.txt'];
  // En seco el fichero puede no estar en disco todavia porque lo trae el andamio en esta misma pasada: se evalua
  // el del andamio, para que el simulacro diga lo mismo que la pasada real (ADR-036 D4).
  const origen = fs.existsSync(dst) ? dst : (DRY && previstos.has('.gitignore') ? path.join(SCAFFOLD, '.gitignore') : null);
  if (origen) {
    const actual = fs.readFileSync(origen, 'utf8');
    const lineas = actual.split(/\r?\n/).map((l) => l.trim());
    const faltan = necesarias.filter((e) => !lineas.includes(e));
    if (faltan.length) {
      if (!DRY) {
        const sep = actual.endsWith('\n') ? '' : '\n';
        fs.appendFileSync(dst, sep + '\n# Ovillo - identidad y estado local (no commitear)\n' + faltan.join('\n') + '\n', 'utf8');
      }
      log.push('  ~ .gitignore: ' + faltan.length + ' entrada(s) local(es) anadida(s) (' + faltan.join(', ') + ')');
      hecho.files++;
    }
  }
}

// 5) IDE: solo si se pide o esta en ecosystem.config.json
let ide = IDE;
if (!ide) { try { ide = JSON.parse(fs.readFileSync(path.join(DEST, 'ecosystem.config.json'), 'utf8')).organization.ide || 'none'; } catch { ide = 'none'; } }
if (ide === 'vscode') copyTree('.vscode');
// El script vive en scripts/ en el origen y en el destino: una sola ruta que citar (ADR-028).
if (ide === 'visual-studio') copy('.claude/scripts/integracion-vs.ps1');

// 6) resumen
const modo = (DRY ? 'DRY-RUN (nada escrito)' : 'hecho') + (REFRESH ? ' con --refresh-memoria' : '');
console.log('[hv] init ' + modo + ' en ' + DEST);
console.log('  carpetas creadas: ' + hecho.dirs + ' · ficheros escritos: ' + hecho.files + ' · conservados (ya existian): ' + hecho.kept + ' · ide: ' + ide);
if (log.length) console.log(log.slice(0, 40).join('\n') + (log.length > 40 ? '\n  ... y ' + (log.length - 40) + ' mas' : ''));
console.log('');
// ADR-023: init nunca reescribe el Hilo; solo avisa si el ESTADO_PROYECTO.json existente conserva claves planas deprecadas
try {
  const e = JSON.parse(fs.readFileSync(path.join(DEST, '_hilo', 'ESTADO_PROYECTO.json'), 'utf8'));
  const planas = ['convencionRamas', 'ramaBase'].filter((k) => e && e.configuracion && e.configuracion[k] !== undefined);
  // La ruta del script depende del modo de distribucion (ADR-035): en modo plugin el proyecto no tiene .claude/hooks/.
  const cmdMigrar = PLUGIN_ROOT ? 'node "' + PLUGIN_ROOT.replace(/\\/g, '/') + '/hooks/config-check.js" --migrate' : 'node .claude/hooks/config-check.js --migrate';
  if (planas.length) console.log('Aviso: _hilo/ESTADO_PROYECTO.json conserva claves planas deprecadas (' + planas.map((k) => 'configuracion.' + k).join(', ') + '); migra con: ' + cmdMigrar);
} catch { /* sin Hilo o ilegible: nada que decir */ }
console.log('Siguiente: copia ecosystem.config.example.json a ecosystem.config.json y rellena organization; luego /hv:onboarding.');
