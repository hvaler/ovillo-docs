// TEMPLATE: API Configuration Options
// Uso: Clases de configuración para APIs externas con IOptions pattern

using System.ComponentModel.DataAnnotations;

namespace MyCompany.MyApp.Infrastructure.ApiClients.Configuration;

/// <summary>
/// Configuración base para cualquier API externa
/// </summary>
public abstract class ApiOptionsBase
{
    [Required(ErrorMessage = "BaseUrl is required")]
    [Url(ErrorMessage = "BaseUrl must be a valid URL")]
    public string BaseUrl { get; set; } = string.Empty;

    [Range(5, 300, ErrorMessage = "TimeoutSeconds must be between 5 and 300")]
    public int TimeoutSeconds { get; set; } = 30;

    [Range(0, 10, ErrorMessage = "MaxRetries must be between 0 and 10")]
    public int MaxRetries { get; set; } = 3;

    public bool EnableLogging { get; set; } = true;
}

/// <summary>
/// Configuración para SistemaA API
/// </summary>
public class SistemaAApiOptions : ApiOptionsBase
{
    public const string SectionName = "SistemaAApi";

    [Required(ErrorMessage = "ClientId is required")]
    public string ClientId { get; set; } = string.Empty;

    [Required(ErrorMessage = "ClientSecret is required")]
    public string ClientSecret { get; set; } = string.Empty;

    [Required(ErrorMessage = "TokenUrl is required")]
    [Url(ErrorMessage = "TokenUrl must be a valid URL")]
    public string TokenUrl { get; set; } = string.Empty;

    public string? Scope { get; set; }

    public int TokenCacheDurationMinutes { get; set; } = 50; // 90% de 60 minutos
}

/// <summary>
/// Configuración para Oracle HCM Cloud API
/// </summary>
public class SistemaBApiOptions : ApiOptionsBase
{
    public const string SectionName = "SistemaBApi";

    [Required(ErrorMessage = "ClientId is required")]
    public string ClientId { get; set; } = string.Empty;

    [Required(ErrorMessage = "ClientSecret is required")]
    public string ClientSecret { get; set; } = string.Empty;

    [Required(ErrorMessage = "TokenUrl is required")]
    [Url]
    public string TokenUrl { get; set; } = string.Empty;

    public int PageSize { get; set; } = 100;

    public int CacheDurationMinutes { get; set; } = 60;
}

/// <summary>
/// Configuración para SistemaC API
/// </summary>
public class SistemaCApiOptions : ApiOptionsBase
{
    public const string SectionName = "SistemaCApi";

    [Required(ErrorMessage = "ApiKey is required")]
    public string ApiKey { get; set; } = string.Empty;

    [Required(ErrorMessage = "ApiSecret is required")]
    public string ApiSecret { get; set; } = string.Empty;

    public bool UseHmacAuthentication { get; set; } = true;
}

// ============================================================================
// Extension Methods para Registro en DI
// ============================================================================

public static class ApiOptionsExtensions
{
    /// <summary>
    /// Registra opciones de SistemaA API con validación
    /// </summary>
    public static IServiceCollection AddSistemaAApiOptions(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        services.AddOptionsWithValidateOnStart<SistemaAApiOptions>()
            .Bind(configuration.GetSection(SistemaAApiOptions.SectionName))
            .ValidateDataAnnotations()
            .Validate(options =>
            {
                // Validaciones adicionales
                if (options.TimeoutSeconds > 120)
                {
                    Console.WriteLine("⚠️ Warning: TimeoutSeconds > 120s puede causar problemas de rendimiento");
                }

                return true;
            });

        return services;
    }

    /// <summary>
    /// Registra opciones de Oracle HCM API con validación
    /// </summary>
    public static IServiceCollection AddSistemaBApiOptions(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        services.AddOptionsWithValidateOnStart<SistemaBApiOptions>()
            .Bind(configuration.GetSection(SistemaBApiOptions.SectionName))
            .ValidateDataAnnotations();

        return services;
    }

    /// <summary>
    /// Registra opciones de SistemaC API con validación
    /// </summary>
    public static IServiceCollection AddSistemaCApiOptions(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        services.AddOptionsWithValidateOnStart<SistemaCApiOptions>()
            .Bind(configuration.GetSection(SistemaCApiOptions.SectionName))
            .ValidateDataAnnotations();

        return services;
    }
}
