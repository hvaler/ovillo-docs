    @* Script de layout - Sidebar y navegación *@
    <script>
        // Namespace de la app
        const App = window.App || {};

        App.Layout = (() => {
            'use strict';

            const elements = {
                sidebar: null,
                sidebarToggle: null,
                hamburger: null,
                overlay: null
            };

            const init = () => {
                elements.sidebar = document.querySelector('.sidebar');
                elements.sidebarToggle = document.querySelector('.sidebar-toggle');
                elements.hamburger = document.querySelector('.hamburger');
                elements.overlay = document.getElementById('sidebarOverlay');

                bindEvents();
                initSubmenus();
            };

            const bindEvents = () => {
                // Toggle sidebar (desktop/tablet)
                if (elements.sidebarToggle) {
                    elements.sidebarToggle.addEventListener('click', toggleSidebar);
                }

                // Hamburger menu (móvil)
                if (elements.hamburger) {
                    elements.hamburger.addEventListener('click', toggleMobileSidebar);
                }

                // Cerrar sidebar al click en overlay
                if (elements.overlay) {
                    elements.overlay.addEventListener('click', closeMobileSidebar);
                }

                // Cerrar con Escape
                document.addEventListener('keydown', (e) => {
                    if (e.key === 'Escape' && elements.sidebar?.classList.contains('open')) {
                        closeMobileSidebar();
                    }
                });
            };

            const toggleSidebar = () => {
                document.body.classList.toggle('sidebar-collapsed');

                // Guardar preferencia
                const isCollapsed = document.body.classList.contains('sidebar-collapsed');
                localStorage.setItem('sidebarCollapsed', isCollapsed);

                // Actualizar aria-expanded
                if (elements.sidebarToggle) {
                    elements.sidebarToggle.setAttribute('aria-expanded', !isCollapsed);
                }
            };

            const toggleMobileSidebar = () => {
                elements.sidebar?.classList.toggle('open');
                elements.hamburger?.classList.toggle('open');

                // Gestionar focus trap
                if (elements.sidebar?.classList.contains('open')) {
                    elements.sidebar.querySelector('a, button')?.focus();
                }
            };

            const closeMobileSidebar = () => {
                elements.sidebar?.classList.remove('open');
                elements.hamburger?.classList.remove('open');
                elements.hamburger?.focus();
            };

            const initSubmenus = () => {
                const submenuButtons = document.querySelectorAll('.nav-item.has-submenu > .nav-link');

                submenuButtons.forEach(btn => {
                    btn.addEventListener('click', (e) => {
                        e.preventDefault();
                        const parent = btn.parentElement;
                        const isExpanded = btn.getAttribute('aria-expanded') === 'true';

                        // Cerrar otros submenús
                        submenuButtons.forEach(other => {
                            if (other !== btn) {
                                other.setAttribute('aria-expanded', 'false');
                                other.parentElement?.classList.remove('expanded');
                            }
                        });

                        // Toggle actual
                        btn.setAttribute('aria-expanded', !isExpanded);
                        parent?.classList.toggle('expanded');
                    });
                });
            };

            // Restaurar estado del sidebar
            const restoreState = () => {
                const isCollapsed = localStorage.getItem('sidebarCollapsed') === 'true';
                if (isCollapsed) {
                    document.body.classList.add('sidebar-collapsed');
                }
            };

            // Inicializar al cargar
            restoreState();

            return { init, toggleSidebar, toggleMobileSidebar, closeMobileSidebar };
        })();

        document.addEventListener('DOMContentLoaded', App.Layout.init);
    </script>

    @* Scripts adicionales de la página *@
    @await RenderSectionAsync("Scripts", required: false)
</body>
